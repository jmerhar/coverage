1"""The API access policy, in one place. 

2 

3This module is the *whole* backend permission policy. It is deliberately 

4fail-closed: an endpoint is admin-only unless it is listed below. Adding a route 

5without touching this file makes it admin-only, and ``test_access.py`` fails 

6until the omission is made deliberate — the opposite of an opt-in decorator, 

7where a forgotten dependency silently ships a public mutation. 

8 

9Keys are ``(method, full-path-template)``, e.g. 

10``("GET", "/api/properties/{property_id}")``. ``enforce_access`` is mounted once 

11as a global dependency in :mod:`idealista.api.app`; individual routers carry no 

12auth dependencies of their own. 

13 

14See ``docs/permissions.md`` for the access model and how to add an endpoint. 

15""" 

16 

17from fastapi import HTTPException, Request 

18 

19from idealista.api.deps import ADMIN_COOKIE, is_admin_token 

20from idealista.config import API_PREFIX 

21 

22#: Re-exported so readers of the declarations below — which all include the prefix — 

23#: can see where it comes from, and so :mod:`idealista.api.app` mounts the routers 

24#: at the same prefix this policy is written against. 

25__all__ = ["API_PREFIX", "PUBLIC", "TOKEN_AUTHED", "enforce_access", "is_open"] 

26 

27#: Endpoints reachable with no credentials at all. 

28#: 

29#: Scoped to what the pages a logged-out visitor can reach actually fetch: 

30#: Dashboard, Properties (+ detail), Scrape Targets (read-only), Filters 

31#: (view + apply) and Settings (view). Everything else is admin-only. 

32PUBLIC: frozenset[tuple[str, str]] = frozenset( 

33 { 

34 # --- Infrastructure and the login gate itself --- 

35 ("GET", "/api/health"), 

36 ("GET", "/api/auth/me"), 

37 ("POST", "/api/auth/login"), 

38 ("POST", "/api/auth/logout"), 

39 # --- Server-rendered OG tags --- 

40 # nginx proxies *every* bare /properties/{id} request here, not just 

41 # crawler traffic (frontend/nginx.conf), so gating this would break both 

42 # social link previews and every property detail page load. 

43 ("GET", "/api/seo/properties/{property_id}"), 

44 # --- Dashboard --- 

45 ("GET", "/api/dashboard"), 

46 ("GET", "/api/scrape-targets/status"), 

47 # --- Properties list and detail --- 

48 ("GET", "/api/properties"), 

49 ("GET", "/api/properties/{property_id}"), 

50 # Backs the Translate button on the detail page. The only public 

51 # endpoint that persists anything: it writes a file-based translation 

52 # cache and calls the translation API. No database write. 

53 ("GET", "/api/properties/{property_id}/translation"), 

54 # Read by the tag filter in the Properties filter panel, so it stays 

55 # public even though the Tags *page* is admin-only. 

56 ("GET", "/api/tags"), 

57 # Location reference data feeding the location filter. 

58 ("GET", "/api/locations/districts"), 

59 ("GET", "/api/locations/districts/{district_id}/municipalities"), 

60 ("GET", "/api/locations/municipalities/{municipality_id}/parishes"), 

61 ("GET", "/api/locations/tree"), 

62 # Stateless what-if computation behind the yield calculator: no DB, no 

63 # persistence, despite being a POST. 

64 ("POST", "/api/yield/calculate"), 

65 # --- Scrape Targets, read-only --- 

66 ("GET", "/api/scrape-targets"), 

67 ("GET", "/api/scrape-targets/{target_id}"), 

68 # --- Saved filters: view and apply, but no create/edit/delete --- 

69 ("GET", "/api/saved-filters"), 

70 # --- Settings: view the values, but no save --- 

71 ("GET", "/api/settings/yield"), 

72 } 

73)

75#: Endpoints whose credential is the unguessable token in their own URL. 

76#: 

77#: These are reached from captcha notifications by whoever holds the link, which 

78#: is why they take no cookie. The token is validated by the handlers themselves 

79#: (unknown -> 404, expired or already used -> 410). 

80TOKEN_AUTHED: frozenset[tuple[str, str]] = frozenset( 

81 { 

82 ("GET", "/api/captcha-solve/{token}"), 

83 ("POST", "/api/captcha-solve/{token}"), 

84 ("GET", "/api/captcha-solve/{token}/resume"), 

85 ("POST", "/api/captcha-solve/{token}/resume"), 

86 } 

87) 

88 

89 

90#: Everything reachable without an admin cookie. 

91_OPEN = PUBLIC | TOKEN_AUTHED

94def route_templates(app) -> set[tuple[str, str]]: 

95 """Return every ``(method, full-path-template)`` the app serves. 

96 

97 Read from ``app.openapi()``, which reports full templates identically across 

98 FastAPI versions — unlike the router's own route objects, whose paths are 

99 relative to their mount point under some versions and absolute under others. 

100 Used by the tests to hold the declarations to the real route table. 

101 """ 

102 return { 

103 (method.upper(), path) 

104 for path, operations in app.openapi()["paths"].items() 

105 for method in operations 

106 } 

107 

108 

109def candidate_templates(route_path: str) -> tuple[str, ...]: 

110 """Return the full-path templates a matched route's own path could mean. 

111 

112 A route reports its path relative to where it was mounted under some FastAPI 

113 versions (``/health``) and in full under others (``/api/health``), and that has 

114 changed across minor releases. Rather than depend on which, both readings are 

115 checked against the declarations: the mounted prefix is ours, so prepending it 

116 is unambiguous, and a wrong guess simply fails to match. 

117 """ 

118 return (route_path, API_PREFIX + route_path) 

119 

120 

121def is_open(method: str, route_path: str) -> bool: 

122 """Return whether the matched route is reachable without an admin cookie.""" 

123 return any((method, template) in _OPEN for template in candidate_templates(route_path))

126async def enforce_access(request: Request) -> None: 

127 """Apply the access policy to every API request. 

128 

129 Mounted once as a global dependency, so it runs before any endpoint. Anything 

130 not declared open needs a valid admin cookie. Keying off the route the router 

131 actually matched — rather than pattern-matching the requested URL — means a 

132 declaration cannot accidentally cover a neighbouring endpoint. 

133 

134 A request that matched no route has no declaration and so needs the cookie. In 

135 practice that cannot happen: unmatched paths 404 before dependencies run, and 

136 FastAPI's own ``/api/docs`` and ``/api/openapi.json`` are plain Starlette routes, 

137 which do not execute dependencies at all. 

138 """ 

139 route_path = getattr(request.scope.get("route"), "path", None) 

140 if route_path is not None and is_open(request.method, route_path): 

141 return 

142 if not is_admin_token(request.cookies.get(ADMIN_COOKIE)): 

143 raise HTTPException(status_code=403, detail="Admin access required")
