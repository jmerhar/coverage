1"""Captcha-event recording — writes one row per captcha / protection encounter. 

2 

3Used by both the live scraper (as challenges are handled) and the one-time log 

4backfill. Both compute the same ``dedup_key`` and insert with ``ON CONFLICT DO 

5NOTHING`` so the two sources never double-count and re-runs are idempotent.

22def make_dedup_key( 

23 occurred_at: datetime, kind: str, outcome: str, target_label: str | None 

24) -> str: 

25 """Build the idempotency key shared by live and backfill inserts. 

26 

27 Second-resolution timestamp + classification + target. Two genuinely 

28 distinct encounters within the same second, kind, outcome and target are 

29 treated as one — an acceptable trade for making backfill safely re-runnable 

30 and impossible to double-count against live rows. 

31 """ 

32 return ( 

33 f"{occurred_at:%Y-%m-%dT%H:%M:%S}|{kind}|{outcome}|{target_label or ''}" 

34 )

37async def record_captcha_event( 

38 session: AsyncSession, 

39 *, 

40 occurred_at: datetime, 

41 kind: str, 

42 outcome: str, 

43 context: str | None = None, 

44 resolve_seconds: float | None = None, 

45 scrape_type: str | None = None, 

46 target_label: str | None = None, 

47 captcha_url: str | None = None, 

48 backoff_seconds: int | None = None, 

49 backoff_until: datetime | None = None, 

50 resumed_at: datetime | None = None, 

51 scrape_run_id: int | None = None, 

52 source: str = "live", 

53) -> None: 

54 """Insert a captcha event, ignoring a duplicate ``dedup_key``. 

55 

56 Args mirror the ``captcha_events`` columns; see the model for their meaning. 

57 ``kind`` is one of ``auto``/``slider``/``hard_block``/``block``/``unknown``; 

58 ``outcome`` one of ``auto_resolved``/``manual_solved``/``manual_timeout``/ 

59 ``backoff``. The insert commits its own transaction. 

60 """ 

61 # Guard the column width so an unusually long label can't fail the insert 

62 # (recording is best-effort and must not be lost). 

63 target_label = target_label[:200] if target_label else target_label 

64 dedup_key = make_dedup_key(occurred_at, kind, outcome, target_label) 

65 stmt = ( 

66 pg_insert(CaptchaEvent) 

67 .values( 

68 occurred_at=occurred_at, 

69 kind=kind, 

70 context=context, 

71 outcome=outcome, 

72 resolve_seconds=resolve_seconds, 

73 scrape_type=scrape_type, 

74 target_label=target_label, 

75 captcha_url=captcha_url, 

76 backoff_seconds=backoff_seconds, 

77 backoff_until=backoff_until, 

78 resumed_at=resumed_at, 

79 scrape_run_id=scrape_run_id, 

80 source=source, 

81 dedup_key=dedup_key, 

82 ) 

83 .on_conflict_do_nothing(index_elements=["dedup_key"]) 

84 ) 

85 await session.execute(stmt) 

86 await session.commit()

89async def mark_latest_resumed( 

90 session: AsyncSession, resumed_at: datetime 

91) -> bool: 

92 """Record that the operator ended a cooldown early via the resume link. 

93 

94 Sets ``resumed_at`` on the most recent event whose back-off is still active 

95 at ``resumed_at`` (``backoff_until >= resumed_at``) and isn't already marked 

96 resumed, so the dashboard can show the *actual* (shortened) back-off. The 

97 ``backoff_until >= resumed_at`` bound ensures a resume is attributed to the 

98 cooldown actually being ended, never to a stale/historical back-off event 

99 (e.g. one that recorded no live row). Returns True if a row was updated. 

100 """ 

101 target_id = await session.scalar( 

102 select(CaptchaEvent.id) 

103 .where( 

104 CaptchaEvent.backoff_until.is_not(None), 

105 CaptchaEvent.backoff_until >= resumed_at, 

106 CaptchaEvent.resumed_at.is_(None), 

107 ) 

108 .order_by(CaptchaEvent.occurred_at.desc(), CaptchaEvent.id.desc()) 

109 ) 

110 if target_id is None: 

111 return False 

112 await session.execute( 

113 update(CaptchaEvent) 

114 .where(CaptchaEvent.id == target_id) 

115 .values(resumed_at=resumed_at) 

116 ) 

117 await session.commit() 

118 return True
