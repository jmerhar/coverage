/** Captcha page — DataDome captcha behaviour over time (admin-only).
 *
 * Shows how captchas evolved: kind (automatic / slider / hard block / block)
 * over time, breakdowns by scrape type and UTC hour, a drill-down event table,
 * and the current + past captcha settings. Supports trailing presets, prev/next
 * paging, and an absolute custom window — all reflected in the URL so a view is
 * shareable.
 */
 
import { useMemo, useState } from "react";
import { useSearchParams } from "react-router-dom";
import { ChevronLeft, ChevronRight } from "lucide-react";
import { usePageTitle } from "../hooks/usePageTitle";
import { useCaptchaMetrics } from "../hooks/useCaptchaMetrics";
import CaptchaCharts from "../components/captcha/CaptchaCharts";
import CaptchaEventsTable from "../components/captcha/CaptchaEventsTable";
import CaptchaSettingsPanel from "../components/captcha/CaptchaSettingsPanel";
import {
  canMoveForward,
  currentWindow,
  matchingPreset,
  shiftWindow,
} from "../lib/metrics-window";
import type { MetricsQuery, MetricsRange } from "../types/scraper-metrics";
 
const RANGES: MetricsRange[] = ["1h", "6h", "24h", "7d", "30d"];
 
/** Format an epoch-second value for a <input type="datetime-local"> (local tz). */
function toLocalInput(epochSec: number): string {
  const d = new Date(epochSec * 1000);
  const pad = (n: number) => String(n).padStart(2, "0");
  return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())}T${pad(d.getHours())}:${pad(d.getMinutes())}`;
}
 
/** Human-readable label for the shown window (dates for multi-day, times otherwise). */
function fmtWindowLabel(startSec: number, endSec: number): string {
  const s = new Date(startSec * 1000);
  const e = new Date(endSec * 1000);
  if (endSec - startSec > 2 * 86400) {
    const o: Intl.DateTimeFormatOptions = { day: "numeric", month: "short" };
    return `${s.toLocaleDateString("pt-PT", o)} – ${e.toLocaleDateString("pt-PT", o)}`;
  }
  const d: Intl.DateTimeFormatOptions = {
    day: "numeric",
    month: "short",
    hour: "2-digit",
    minute: "2-digit",
  };
  const t: Intl.DateTimeFormatOptions = { hour: "2-digit", minute: "2-digit" };
  const sameDay = s.toDateString() === e.toDateString();
  return sameDay
    ? `${s.toLocaleString("pt-PT", d)} – ${e.toLocaleTimeString("pt-PT", t)}`
    : `${s.toLocaleString("pt-PT", d)} – ${e.toLocaleString("pt-PT", d)}`;
}
 
/** Parse the range + absolute window from URL search params. */
function parseParams(params: URLSearchParams): {
  range: MetricsRange;
  custom: { after: number; before: number } | null;
} {
  const rawRange = params.get("range");
  const range = (RANGES as string[]).includes(rawRange ?? "")
    ? (rawRange as MetricsRange)
    : "24h";
  const after = Number(params.get("after"));
  const before = Number(params.get("before"));
  const custom =
    Number.isFinite(after) && Number.isFinite(before) && after > 0 && after < before
      ? { after, before }
      : null;
  return { range, custom };
}
 
export default function CaptchaPage() {
  usePageTitle("Captcha");
 
  const [searchParams, setSearchParams] = useSearchParams();
  const { range, custom } = parseParams(searchParams);
 
  const [draft, setDraft] = useState<{ after: string; before: string } | null>(null);
  const [showDeployments, setShowDeployments] = useState(true);
  const [showSettings, setShowSettings] = useState(true);
 
  const query: MetricsQuery = useMemo(
    () => (custom ? { range, after: custom.after, before: custom.before } : { range }),
    [range, custom],
  );
  const { data, isLoading, isError } = useCaptchaMetrics(query);
 
  const nowSec = Math.floor(Date.now() / 1000);
  const win = currentWindow(range, custom, nowSec);
  const span = win.before - win.after;
  const activePreset = matchingPreset(span);
  const canForward = canMoveForward(custom, nowSec);
 
  /** Switch to a trailing preset (drops any absolute window). */
  function selectPreset(r: MetricsRange) {
    setSearchParams({ range: r });
    setDraft(null);
  }
 
  /** Page the window one full span older (−1) or newer (+1). */
  function shift(dir: -1 | 1) {
    const next = shiftWindow(range, custom, dir, Math.floor(Date.now() / 1000));
    if (next) {
      setSearchParams({ range, after: String(next.after), before: String(next.before) });
    } else {
      setSearchParams({ range }); // caught up to now → live
    }
    setDraft(null);
  }
 
  /** Open the custom-window editor, seeded with the currently-shown window. */
  function openCustom() {
    setDraft({ after: toLocalInput(win.after), before: toLocalInput(win.before) });
  }
 
  function applyCustom() {
    if (!draft) return;
    const after = Math.floor(new Date(draft.after).getTime() / 1000);
    const before = Math.floor(new Date(draft.before).getTime() / 1000);
    if (Number.isFinite(after) && Number.isFinite(before) && after < before) {
      setSearchParams({ range, after: String(after), before: String(before) });
      setDraft(null);
    }
  }
 
  return (
    <div className="space-y-6">
      <div className="flex flex-wrap items-center justify-between gap-3">
        <h2 className="text-2xl font-bold text-gray-900 dark:text-gray-100">Captcha</h2>
 
        <div className="flex flex-wrap items-center gap-3">
          <div className="flex gap-1 rounded-lg border border-gray-200 p-1 dark:border-gray-700">
            {RANGES.map((r) => (
              <button
                key={r}
                onClick={() => selectPreset(r)}
                className={`rounded-md px-3 py-1 text-sm font-medium transition-colors ${
                  r === activePreset
                    ? "bg-gray-900 text-white dark:bg-gray-100 dark:text-gray-900"
                    : "text-gray-600 hover:bg-gray-100 dark:text-gray-400 dark:hover:bg-gray-800"
                }`}
              >
                {r}
              </button>
            ))}
            <button
              onClick={openCustom}
              className={`rounded-md px-3 py-1 text-sm font-medium transition-colors ${
                custom && activePreset === null
                  ? "bg-gray-900 text-white dark:bg-gray-100 dark:text-gray-900"
                  : "text-gray-600 hover:bg-gray-100 dark:text-gray-400 dark:hover:bg-gray-800"
              }`}
            >
              Custom
            </button>
          </div>
 
          <div className="flex items-center gap-1 rounded-lg border border-gray-200 p-1 dark:border-gray-700">
            <button
              onClick={() => shift(-1)}
              aria-label="Earlier window"
              title="Earlier"
              className="rounded-md p-1 text-gray-600 hover:bg-gray-100 dark:text-gray-400 dark:hover:bg-gray-800"
            >
              <ChevronLeft className="h-4 w-4" />
            </button>
            <span className="min-w-[9rem] text-center text-xs font-medium text-gray-600 dark:text-gray-400">
              {fmtWindowLabel(win.after, win.before)}
            </span>
            <button
              onClick={() => shift(1)}
              disabled={!canForward}
              aria-label="Later window"
              title="Later"
              className="rounded-md p-1 text-gray-600 enabled:hover:bg-gray-100 disabled:opacity-30 dark:text-gray-400 dark:enabled:hover:bg-gray-800"
            >
              <ChevronRight className="h-4 w-4" />
            </button>
          </div>
 
          <div className="flex items-center gap-3 text-sm text-gray-700 dark:text-gray-300">
            <label className="flex items-center gap-1">
              <input
                type="checkbox"
                checked={showDeployments}
                onChange={(e) => setShowDeployments(e.target.checked)}
              />
              Deployments
            </label>
            <label className="flex items-center gap-1">
              <input
                type="checkbox"
                checked={showSettings}
                onChange={(e) => setShowSettings(e.target.checked)}
              />
              Setting changes
            </label>
          </div>
        </div>
      </div>
 
      {draft && (
        <div className="flex flex-wrap items-end gap-3 rounded-lg border border-gray-200 p-3 dark:border-gray-700">
          <label className="text-sm text-gray-700 dark:text-gray-300">
            From
            <input
              type="datetime-local"
              value={draft.after}
              onChange={(e) => setDraft({ ...draft, after: e.target.value })}
              className="ml-2 rounded border border-gray-300 bg-white px-2 py-1 text-sm dark:border-gray-600 dark:bg-gray-800"
            />
          </label>
          <label className="text-sm text-gray-700 dark:text-gray-300">
            To
            <input
              type="datetime-local"
              value={draft.before}
              onChange={(e) => setDraft({ ...draft, before: e.target.value })}
              className="ml-2 rounded border border-gray-300 bg-white px-2 py-1 text-sm dark:border-gray-600 dark:bg-gray-800"
            />
          </label>
          <button
            onClick={applyCustom}
            className="rounded-md bg-gray-900 px-3 py-1 text-sm font-medium text-white dark:bg-gray-100 dark:text-gray-900"
          >
            Apply
          </button>
        </div>
      )}
 
      {isLoading && <p className="text-sm text-gray-500 dark:text-gray-400">Loading captcha metrics…</p>}
      {isError && <p className="text-sm text-red-600 dark:text-red-400">Failed to load captcha metrics.</p>}
 
      {data && (
        <>
          <CaptchaCharts
            buckets={data.buckets}
            summary={data.summary}
            byScrapeType={data.by_scrape_type}
            byHour={data.by_hour_utc}
            workSeries={data.work_series}
            workAvgSeries={data.work_avg_series}
            deployMarkers={data.deploy_markers}
            settingsChanges={data.settings_changes}
            windowStart={data.window_start}
            windowEnd={data.window_end}
            showDeployments={showDeployments}
            showSettings={showSettings}
          />
 
          <section>
            <h3 className="mb-2 text-sm font-semibold text-gray-700 dark:text-gray-300">
              Captcha settings
            </h3>
            <CaptchaSettingsPanel current={data.current_settings} history={data.settings_changes} />
          </section>
 
          <section>
            <h3 className="mb-2 text-sm font-semibold text-gray-700 dark:text-gray-300">
              Events
            </h3>
            <CaptchaEventsTable events={data.events} />
          </section>
        </>
      )}
    </div>
  );
}
