/** Presentational charts + summary for the captcha dashboard.
 *
 * Pure/visual only — all data is passed in, so it renders in tests without
 * network. CaptchaPage owns data fetching, window state, and toggles.
 */
 
import type { ReactElement, ReactNode } from "react";
import { useMemo, useState } from "react";
import {
  Area,
  Bar,
  BarChart,
  CartesianGrid,
  ComposedChart,
  Legend,
  Line,
  ReferenceLine,
  ResponsiveContainer,
  Scatter,
  Tooltip,
  XAxis,
  YAxis,
} from "recharts";
import type {
  CaptchaBucket,
  CaptchaSummary,
  DeployMarker,
  HourCount,
  ScrapeTypeCount,
  SettingsChange,
  WorkAvgPoint,
  WorkPoint,
} from "../../types/captcha-metrics";
 
type KindKey = "auto" | "slider" | "hard_block" | "block";
 
const COLORS = {
  auto: "#22c55e", // green-500 — benign auto-resolve
  slider: "#f59e0b", // amber-500 — needed a manual solve
  hard_block: "#ef4444", // red-500 — unsolvable t=bv
  block: "#a855f7", // purple-500 — 403/blocked
  deploy: "#ef4444", // red-500 (dashed)
  setting: "#14b8a6", // teal-500 (dashed)
} as const;
 
const KINDS: { key: KindKey; label: string }[] = [
  { key: "auto", label: "Automatic" },
  { key: "slider", label: "Slider" },
  { key: "hard_block", label: "Hard block" },
  { key: "block", label: "Block" },
];
 
const epoch = (iso: string) => Math.floor(new Date(iso).getTime() / 1000);
 
/** Format seconds compactly (s / m / h). */
function fmtDuration(secs: number | null): string {
  if (secs == null) return "—";
  if (secs < 90) return `${Math.round(secs)}s`;
  if (secs < 5400) return `${(secs / 60).toFixed(1)}m`;
  return `${(secs / 3600).toFixed(1)}h`;
}
 
/** Build an epoch-second tick formatter; uses dates for multi-day windows. */
function makeTimeFormatter(spanSecs: number) {
  const multiDay = spanSecs > 2 * 86400;
  return (t: number) => {
    const d = new Date(t * 1000);
    return multiDay
      ? d.toLocaleDateString("pt-PT", { day: "numeric", month: "short" })
      : d.toLocaleTimeString("pt-PT", { hour: "2-digit", minute: "2-digit" });
  };
}
 
interface ChartCardProps {
  title: string;
  children: ReactElement;
  height?: number;
  headerRight?: ReactNode;
}
 
function ChartCard({ title, children, height = 280, headerRight }: ChartCardProps) {
  return (
    <section>
      <div className="mb-2 flex items-center justify-between gap-3">
        <h3 className="text-sm font-semibold text-gray-700 dark:text-gray-300">{title}</h3>
        {headerRight}
      </div>
      <div className="rounded-lg border border-gray-200 p-3 dark:border-gray-700">
        <ResponsiveContainer width="100%" height={height}>
          {children}
        </ResponsiveContainer>
      </div>
    </section>
  );
}
 
/** Nearest-rank percentile of an unsorted number array (0 if empty). */
function percentile(values: number[], p: number): number {
  if (values.length === 0) return 0;
  const sorted = [...values].sort((a, b) => a - b);
  const idx = Math.min(sorted.length - 1, Math.max(0, Math.ceil(p * sorted.length) - 1));
  return sorted[idx];
}
 
/** Round up to a clean axis bound (nearest 1k above 2k, else nearest 100). */
function niceCap(v: number): number {
  if (v >= 2000) return Math.ceil(v / 1000) * 1000;
  if (v >= 200) return Math.ceil(v / 100) * 100;
  return Math.max(Math.ceil(v / 10) * 10, 10);
}
 
function SummaryCard({ label, value, hint }: { label: string; value: string; hint?: string }) {
  return (
    <div className="rounded-lg border border-gray-200 p-3 dark:border-gray-700">
      <div className="text-xs text-gray-500 dark:text-gray-400">{label}</div>
      <div className="mt-1 text-xl font-semibold text-gray-900 dark:text-gray-100">{value}</div>
      {hint && <div className="text-xs text-gray-400">{hint}</div>}
    </div>
  );
}
 
export interface CaptchaChartsProps {
  buckets: CaptchaBucket[];
  summary: CaptchaSummary;
  byScrapeType: ScrapeTypeCount[];
  byHour: HourCount[];
  workSeries: WorkPoint[];
  workAvgSeries: WorkAvgPoint[];
  deployMarkers: DeployMarker[];
  settingsChanges: SettingsChange[];
  windowStart: number; // epoch seconds
  windowEnd: number; // epoch seconds
  showDeployments: boolean;
  showSettings: boolean;
}
 
/** Summary cards + captchas-over-time, by-scrape-type, by-hour, and work-before charts. */
export default function CaptchaCharts({
  buckets,
  summary,
  byScrapeType,
  byHour,
  workSeries,
  workAvgSeries,
  deployMarkers,
  settingsChanges,
  windowStart,
  windowEnd,
  showDeployments,
  showSettings,
}: CaptchaChartsProps) {
  const isDark =
    typeof window !== "undefined" &&
    window.matchMedia("(prefers-color-scheme: dark)").matches;
  const tickColor = isDark ? "#d1d5db" : "#374151";
  const grid = isDark ? "#374151" : "#e5e7eb";
  const fmtTime = makeTimeFormatter(windowEnd - windowStart);
  const tooltipLabel = (t: number) => new Date(t * 1000).toLocaleString("pt-PT");
  const showLabels = deployMarkers.length + settingsChanges.length <= 30;
  const avgColor = isDark ? "#f9fafb" : "#111827"; // neutral, stands out from kinds
  const medianColor = isDark ? "#60a5fa" : "#2563eb"; // blue, distinct from mean + kinds
  const legendStyle = { fontSize: 11 };
 
  // Outlier handling for the pages-since-block chart: a few very long block-free
  // stretches otherwise stretch the axis and squash everything else. Default to
  // clamping the axis near the 98th percentile; a toggle reveals the full range.
  const [clipOutliers, setClipOutliers] = useState(true);
  const workCap = useMemo(
    () => niceCap(percentile(workSeries.map((w) => w.pages), 0.98)),
    [workSeries],
  );
  const clippedCount = clipOutliers
    ? workSeries.filter((w) => w.pages > workCap).length
    : 0;
  const workYDomain: [number, number | string] = clipOutliers
    ? [0, workCap]
    : [0, "auto"];
 
  /** Deploy + settings-change reference lines within the window. */
  function overlays(): ReactElement[] {
    const lines: ReactElement[] = [];
    if (showDeployments) {
      for (const m of deployMarkers) {
        lines.push(
          <ReferenceLine
            key={`d-${m.sha}-${m.at}`}
            x={epoch(m.at)}
            stroke={COLORS.deploy}
            strokeDasharray="3 3"
            strokeOpacity={0.7}
            label={
              showLabels
                ? { value: m.short_sha, fontSize: 9, fill: COLORS.deploy, position: "top" }
                : undefined
            }
          />,
        );
      }
    }
    if (showSettings) {
      for (const s of settingsChanges) {
        const at = epoch(s.effective_from);
        if (at < windowStart || at > windowEnd) continue;
        lines.push(
          <ReferenceLine
            key={`s-${s.effective_from}`}
            x={at}
            stroke={COLORS.setting}
            strokeDasharray="4 2"
            strokeOpacity={0.8}
            label={
              showLabels
                ? { value: "⚙", fontSize: 11, fill: COLORS.setting, position: "top" }
                : undefined
            }
          />,
        );
      }
    }
    return lines;
  }
 
  const xAxisProps = {
    dataKey: "t" as const,
    type: "number" as const,
    domain: [windowStart, windowEnd] as [number, number],
    tickFormatter: fmtTime,
    tick: { fill: tickColor, fontSize: 11 },
    allowDataOverflow: true,
  };
 
  return (
    <div className="space-y-6">
      <section className="grid grid-cols-2 gap-3 sm:grid-cols-3 lg:grid-cols-6">
        <SummaryCard label="Captchas" value={`${summary.captchas}`} hint="slider + hard + block" />
        <SummaryCard label="Auto-resolved" value={`${summary.auto_resolved}`} hint="no escalation" />
        <SummaryCard label="Slider" value={`${summary.slider}`} hint="manual solve needed" />
        <SummaryCard label="Hard blocks" value={`${summary.hard_block}`} hint="t=bv, unsolvable" />
        <SummaryCard label="Blocks" value={`${summary.block}`} hint="403 / rate limit" />
        <SummaryCard
          label="Manual solved"
          value={`${summary.manual_solved}`}
          hint={`${summary.manual_timeout} timed out`}
        />
        <SummaryCard label="Avg auto-resolve" value={fmtDuration(summary.avg_resolve_seconds)} />
        <SummaryCard
          label="Avg pages / block"
          value={summary.avg_pages_since_block != null ? `${summary.avg_pages_since_block}` : "—"}
          hint={
            summary.median_pages_since_block != null
              ? `median ${summary.median_pages_since_block}`
              : undefined
          }
        />
        <SummaryCard
          label="Total back-off"
          value={`${summary.total_backoff_hours}h`}
          hint={summary.avg_backoff_hours != null ? `avg ${summary.avg_backoff_hours}h` : undefined}
        />
        <SummaryCard label="Resumed early" value={`${summary.resumed_count}`} />
        <SummaryCard label="Cold-start" value={`${summary.cold_start_captchas}`} hint="0 pages in" />
        <SummaryCard label="Mid-pass" value={`${summary.mid_pass_captchas}`} hint=">0 pages in" />
      </section>
 
      <ChartCard title="Captchas over time (by kind)">
        <ComposedChart data={buckets} margin={{ top: 16, right: 8, left: 0, bottom: 0 }}>
          <CartesianGrid strokeDasharray="3 3" stroke={grid} />
          <XAxis {...xAxisProps} />
          <YAxis allowDecimals={false} tick={{ fill: tickColor, fontSize: 11 }} />
          <Tooltip labelFormatter={tooltipLabel} />
          <Legend wrapperStyle={legendStyle} />
          {/* Stacked areas (not bars): they render cleanly on the continuous
              numeric time axis at any bucket density and keep the deploy /
              setting-change ReferenceLine overlays aligned. */}
          {KINDS.map((k) => (
            <Area
              key={k.key}
              type="stepAfter"
              dataKey={k.key}
              name={k.label}
              stackId="kind"
              stroke={COLORS[k.key]}
              fill={COLORS[k.key]}
              fillOpacity={0.7}
              isAnimationActive={false}
            />
          ))}
          {overlays()}
        </ComposedChart>
      </ChartCard>
 
      <ChartCard
        title="Pages scraped before each captcha, over time (points by kind, lines = bucket mean & median)"
        headerRight={
          <label className="flex shrink-0 items-center gap-1 text-xs text-gray-600 dark:text-gray-400">
            <input
              type="checkbox"
              checked={clipOutliers}
              onChange={(e) => setClipOutliers(e.target.checked)}
            />
            Clip outliers
            {clippedCount > 0 && (
              <span className="text-gray-400">
                ({clippedCount} above {workCap.toLocaleString("pt-PT")} hidden)
              </span>
            )}
          </label>
        }
      >
        <ComposedChart data={workAvgSeries} margin={{ top: 16, right: 8, left: 0, bottom: 0 }}>
          <CartesianGrid strokeDasharray="3 3" stroke={grid} />
          <XAxis {...xAxisProps} />
          <YAxis
            allowDecimals={false}
            domain={workYDomain}
            allowDataOverflow={clipOutliers}
            tick={{ fill: tickColor, fontSize: 11 }}
          />
          <Tooltip
            labelFormatter={tooltipLabel}
            formatter={(v: number) => `${v} pages`}
          />
          <Legend wrapperStyle={legendStyle} />
          {/* Per-captcha scatter, one series per kind (same colours as above). */}
          {KINDS.map((k) => (
            <Scatter
              key={k.key}
              name={k.label}
              data={workSeries.filter((w) => w.kind === k.key)}
              dataKey="pages"
              fill={COLORS[k.key]}
              isAnimationActive={false}
            />
          ))}
          {/* Bucket mean + median trend lines (mean neutral, median blue). The
              median reads true through the outliers that pull the mean up. */}
          <Line
            type="monotone"
            dataKey="mean"
            name="Average"
            stroke={avgColor}
            strokeWidth={2}
            dot={false}
            isAnimationActive={false}
          />
          <Line
            type="monotone"
            dataKey="median"
            name="Median"
            stroke={medianColor}
            strokeWidth={2}
            dot={false}
            isAnimationActive={false}
          />
          {overlays()}
        </ComposedChart>
      </ChartCard>
 
      <div className="grid gap-6 lg:grid-cols-2">
        <ChartCard title="Captchas by scrape type">
          <BarChart data={byScrapeType} margin={{ top: 8, right: 8, left: 0, bottom: 0 }}>
            <CartesianGrid strokeDasharray="3 3" stroke={grid} />
            <XAxis dataKey="scrape_type" tick={{ fill: tickColor, fontSize: 11 }} />
            <YAxis allowDecimals={false} tick={{ fill: tickColor, fontSize: 11 }} />
            <Tooltip />
            <Legend wrapperStyle={legendStyle} />
            {KINDS.map((k) => (
              <Bar
                key={k.key}
                dataKey={k.key}
                name={k.label}
                stackId="kind"
                fill={COLORS[k.key]}
                isAnimationActive={false}
              />
            ))}
          </BarChart>
        </ChartCard>
 
        <ChartCard title="Captchas by hour of day (UTC)">
          <BarChart data={byHour} margin={{ top: 8, right: 8, left: 0, bottom: 0 }}>
            <CartesianGrid strokeDasharray="3 3" stroke={grid} />
            <XAxis dataKey="hour" tick={{ fill: tickColor, fontSize: 10 }} interval={0} />
            <YAxis allowDecimals={false} tick={{ fill: tickColor, fontSize: 11 }} />
            <Tooltip labelFormatter={(h) => `${h}:00 UTC`} />
            <Legend wrapperStyle={legendStyle} />
            {KINDS.map((k) => (
              <Bar
                key={k.key}
                dataKey={k.key}
                name={k.label}
                stackId="kind"
                fill={COLORS[k.key]}
                isAnimationActive={false}
              />
            ))}
          </BarChart>
        </ChartCard>
      </div>
    </div>
  );
}
