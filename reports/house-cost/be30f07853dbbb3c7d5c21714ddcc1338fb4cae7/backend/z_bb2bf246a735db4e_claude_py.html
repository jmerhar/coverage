82def day_or_none(value: str | None) -> date | None: 

83 """A YYYY-MM-DD date Claude wrote, or nothing when it wrote something else.""" 

84 try: 

85 return date.fromisoformat(value.strip()) if value else None 

86 except ValueError: 

87 return None

90def cents_or_none(value: str | None) -> int | None: 

91 """Euros Claude wrote ("287.59", "€ 1 234,50") as cents, or nothing when unreadable.""" 

92 if not value: 

93 return None 

94 text = re.sub(r"[€\s]", "", value) 

95 if "," in text and "." in text: 

96 # Whichever comes last is the decimal separator; the other groups thousands. 

97 decimal_comma = text.rfind(",") > text.rfind(".") 

98 text = text.replace(".", "").replace(",", ".") if decimal_comma else text.replace(",", "") 

99 else: 

100 text = text.replace(",", ".") 

101 try: 

102 return int((Decimal(text) * 100).quantize(Decimal(1))) 

103 except InvalidOperation: 

104 return None

107def _document(data: bytes, media_type: str, citations: bool = False) -> dict: 

108 """The file as a content block: a PDF as a document, a photo as an image.""" 

109 encoded = base64.standard_b64encode(data).decode() 

110 if media_type == "application/pdf": 

111 block: dict = { 

112 "type": "document", 

113 "source": {"type": "base64", "media_type": media_type, "data": encoded}, 

114 } 

115 if citations: 

116 block["citations"] = {"enabled": True} 

117 return block 

118 source = {"type": "base64", "media_type": media_type, "data": encoded} 

119 return {"type": "image", "source": source}

122class Claude: 

123 """The calls the app makes to Claude, each one request for one file.""" 

124 

125 def __init__( 

126 self, api_key: str, model: str, http_client: httpx2.AsyncClient | None = None 

127 ) -> None: 

128 self.model = model 

129 self.client = anthropic.AsyncAnthropic( 

130 api_key=api_key, timeout=TIMEOUT_SECONDS, http_client=http_client 

131 ) 

132 

133 async def _parse[T: BaseModel](self, content: list[dict], schema: type[T]) -> T: 

134 try: 

135 response = await self.client.messages.parse( 

136 model=self.model, 

137 max_tokens=MAX_TOKENS, 

138 messages=[{"role": "user", "content": content}], 

139 output_format=schema, 

140 ) 

141 except anthropic.APIError as error: 

142 logger.warning("Claude request failed: %s", type(error).__name__) 

143 raise ClaudeFailed("claude_failed") from error 

144 if response.stop_reason == "refusal" or response.parsed_output is None: 

145 raise ClaudeFailed("claude_declined") 

146 return response.parsed_output 

147 

148 async def read_bill(self, data: bytes, media_type: str) -> BillRead: 

149 """The bill's issuer, number, dates, total and parts.""" 

150 return await self._parse( 

151 [ 

152 _document(data, media_type), 

153 { 

154 "type": "text", 

155 "text": "This is a utility or service bill for a house in Portugal. Read " 

156 "what it states; leave out anything it does not show.", 

157 }, 

158 ], 

159 BillRead, 

160 ) 

161 

162 async def read_terms(self, data: bytes, media_type: str, lang: str) -> TermsRead: 

163 """A contract's dates, lock-in and notice, and a summary in ``lang``.""" 

164 return await self._parse( 

165 [ 

166 _document(data, media_type), 

167 { 

168 "type": "text", 

169 "text": "This is a contract, or the terms of one, for a service at a house " 

170 "in Portugal whose costs several households share. Read its dates, any " 

171 "lock-in, the notice needed to leave and how often it bills; leave out " 

172 "what it does not state. Write the summary in " 

173 f"{LANGUAGE_NAMES.get(lang, 'English')}, in a few sentences: the price, " 

174 "the lock-in and what leaving early costs.", 

175 }, 

176 ], 

177 TermsRead, 

178 ) 

179 

180 async def ask(self, data: bytes, media_type: str, question: str, lang: str) -> Answer: 

181 """An answer to ``question`` from the document, citing the pages it rests on.""" 

182 try: 

183 response = await self.client.messages.create( 

184 model=self.model, 

185 max_tokens=MAX_TOKENS, 

186 messages=[ 

187 { 

188 "role": "user", 

189 "content": [ 

190 _document(data, media_type, citations=True), 

191 { 

192 "type": "text", 

193 "text": "Answer from this document only, briefly, in " 

194 f"{LANGUAGE_NAMES.get(lang, 'English')}. If it does not say, " 

195 f"say so.\n\n{question}", 

196 }, 

197 ], 

198 } 

199 ], 

200 ) 

201 except anthropic.APIError as error: 

202 logger.warning("Claude request failed: %s", type(error).__name__) 

203 raise ClaudeFailed("claude_failed") from error 

204 if response.stop_reason == "refusal": 

205 raise ClaudeFailed("claude_declined") 

206 text, citations = [], [] 

207 for block in response.content: 

208 if block.type != "text": 

209 continue 

210 text.append(block.text) 

211 for cited in block.citations or []: 

212 page = getattr(cited, "start_page_number", None) 

213 citations.append(Citation(page=page, quote=cited.cited_text.strip())) 

214 return Answer(text="".join(text).strip(), citations=citations)
