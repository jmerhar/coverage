25async def back_up(engine: AsyncEngine, directory: Path, today: str, keep: int) -> Path | None: 

26 """Copy the database to ``directory`` as today's snapshot, unless there is one, and keep only 

27 the ``keep`` newest. 

28 

29 VACUUM INTO writes a consistent copy while the app goes on writing, which a plain file copy 

30 of a database in use cannot promise. A backup of the data directory then holds a snapshot it 

31 can restore even when it caught the live file mid-write. 

32 """ 

33 directory.mkdir(parents=True, exist_ok=True) 

34 target = directory / f"house-cost-{today}.db" 

35 if target.exists(): 

36 return None 

37 partial = target.with_suffix(".partial") 

38 partial.unlink(missing_ok=True) 

39 async with engine.connect() as connection: 

40 raw = await connection.execution_options(isolation_level="AUTOCOMMIT") 

41 await raw.exec_driver_sql("VACUUM INTO ?", (str(partial),)) 

42 partial.rename(target) 

43 for old in sorted(directory.glob("house-cost-*.db"), reverse=True)[keep:]: 

44 old.unlink() 

45 logger.info("backed the database up to %s", target.name) 

46 return target

49class Scheduler: 

50 """Runs ``tick`` every few minutes for as long as the app runs."""

52 def __init__( 

53 self, 

54 settings: Settings, 

55 engine: AsyncEngine, 

56 sessionmaker: async_sessionmaker[AsyncSession], 

57 notifier: Notifier, 

58 ) -> None: 

59 self.settings = settings 

60 self.engine = engine 

61 self.sessionmaker = sessionmaker 

62 self.notifier = notifier 

63 self._task: asyncio.Task | None = None 

64 self._last_reminded: str | None = None

66 async def tick(self, now: datetime | None = None) -> None: 

67 """Do whatever is due: the day's backup once its time has come, and the day's reminders 

68 once a day.""" 

69 now = now or local_now(self.settings) 

70 today = now.date().isoformat() 

71 hour, minute = (int(part) for part in self.settings.backup_time.split(":")) 

72 if now.time() >= time(hour, minute): 

73 await back_up(self.engine, self.settings.backups_dir, today, self.settings.backups_kept) 

74 if self._last_reminded != today: 

75 async with session_scope(self.sessionmaker) as db: 

76 await contract_reminders(db, self.notifier, now.date()) 

77 await missing_bills(db, self.notifier, now.date()) 

78 self._last_reminded = today 

79 

80 async def _loop(self) -> None: 

81 while True: 

82 try: 

83 await self.tick() 

84 except Exception: # noqa: BLE001 - the next tick tries again 

85 logger.exception("scheduled work failed") 

86 await asyncio.sleep(self.settings.scheduler_tick_seconds) 

87 

88 def start(self) -> None: 

89 """Start the loop, when the settings allow it.""" 

90 if self.settings.scheduler_enabled and self._task is None: 

91 self._task = asyncio.create_task(self._loop(), name="scheduler") 

92 

93 async def stop(self) -> None: 

94 """Stop the loop and wait for it to end.""" 

95 if self._task is None: 

96 return 

97 self._task.cancel() 

98 with contextlib.suppress(asyncio.CancelledError): 

99 await self._task 

100 self._task = None
