1"""Reading text from images with Tesseract, and telling when a PDF's own text cannot be used. 

2 

3Some PDFs carry a text layer in a font whose characters map to nonsense (MEO's telecom invoices 

4read as long runs of "d"), and a scan has none at all; OCR of the rendered page reads both. 

5""" 

6 

7import functools 

8import io 

9import shutil 

10import subprocess 

11 

12from PIL import Image 

13 

14#: How long one page may take to read; a page that takes longer is skipped. 

15PAGE_SECONDS = 60 

16#: The languages read, in order of preference, when Tesseract has their data. 

17LANGUAGES = ("por", "eng") 

18 

19 

20class OcrUnavailable(RuntimeError): 

21 """Tesseract is not installed, so images cannot be read.""" 

22 

23 

24@functools.cache 

25def languages() -> str: 

26 """The `-l` argument for the installed languages among LANGUAGES.""" 

27 binary = shutil.which("tesseract") 

28 if binary is None: 

29 raise OcrUnavailable("tesseract is not installed") 

30 listed = subprocess.run( 

31 [binary, "--list-langs"], capture_output=True, text=True, timeout=30, check=False 

32 ).stdout.split() 

33 present = [language for language in LANGUAGES if language in listed] 

34 return "+".join(present or ["eng"]) 

35 

36 

37def read(image: Image.Image) -> str: 

38 """The text Tesseract reads in ``image``; empty when it reads nothing or gives up.""" 

39 binary = shutil.which("tesseract") 

40 if binary is None: 

41 raise OcrUnavailable("tesseract is not installed") 

42 png = io.BytesIO() 

43 image.convert("L").save(png, format="PNG") 

44 try: 

45 result = subprocess.run( 

46 [binary, "-", "-", "-l", languages(), "--psm", "3"], 

47 input=png.getvalue(), 

48 capture_output=True, 

49 timeout=PAGE_SECONDS, 

50 check=False, 

51 ) 

52 except subprocess.TimeoutExpired: 

53 return "" 

54 return result.stdout.decode("utf-8", errors="replace")

57def unusable(text: str) -> bool: 

58 """Whether a page's text layer is missing or garbled, so the page is better read by OCR. 

59 

60 A page with next to no letters has no text layer, as a scan has none. Garbled text keeps the 

61 letters' count but not their mix: real Portuguese or English has about two vowels in five 

62 letters, while a scrambled font gives hardly any. A short page is judged only once it has 

63 enough letters for that mix to mean something. 

64 """ 

65 letters = [c for c in text.lower() if c.isalpha()] 

66 if len(letters) < 5: 

67 return True 

68 if len(letters) < 40: 

69 return False 

70 vowels = sum(c in "aeiouáàâãéêíóôõú" for c in letters) 

71 return vowels / len(letters) < 0.2
