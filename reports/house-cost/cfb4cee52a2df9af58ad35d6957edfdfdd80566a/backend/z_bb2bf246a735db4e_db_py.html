19def _apply_sqlite_pragmas(dbapi_connection: DBAPIConnection, _record: ConnectionPoolEntry) -> None: 

20 """Turn on the SQLite behaviour this schema assumes. 

21 

22 ``foreign_keys`` is off by default in SQLite, which silently makes every ``ON DELETE`` rule a 

23 no-op. WAL lets the scheduler write while a request reads instead of the two blocking each 

24 other. 

25 """ 

26 cursor = dbapi_connection.cursor() 

27 try: 

28 cursor.execute("PRAGMA foreign_keys=ON") 

29 cursor.execute("PRAGMA journal_mode=WAL") 

30 cursor.execute("PRAGMA busy_timeout=5000") 

31 finally: 

32 cursor.close()
