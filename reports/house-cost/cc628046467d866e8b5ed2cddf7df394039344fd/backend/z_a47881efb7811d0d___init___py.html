34 async def send(self, urls: list[str], title: str, body: str) -> bool:

37 @staticmethod 

38 def _send(urls: list[str], title: str, body: str) -> bool: 

39 import apprise 

40 

41 notifier = apprise.Apprise() 

42 for url in urls: 

43 if not notifier.add(url): 

44 # Logged without the URL, which can carry a password or a token. 

45 logger.warning("an Apprise URL was not recognised and is skipped") 

46 if not len(notifier): 

47 return False 

48 return bool(notifier.notify(body=body, title=title))

51def with_recipient(url: str, email: str) -> str: 

52 """The deployment's mail URL addressed to ``email``, as Apprise's ``to`` argument.""" 

53 parts = urlsplit(url) 

54 to = "to=" + quote(email, safe="@+.") 

55 query = f"{parts.query}&{to}" if parts.query else to 

56 return urlunsplit(parts._replace(query=query))

59def with_target(url: str, phone: str) -> str: 

60 """The deployment's Signal URL sending to ``phone``, the last part of its path.""" 

61 parts = urlsplit(url) 

62 return urlunsplit(parts._replace(path=parts.path.rstrip("/") + "/" + phone))

65@dataclass 

66class Notifier:

73 @property 

74 def channels(self) -> list[str]: 

75 """The kinds of contact people can give: "email", "signal", or neither.""" 

76 offered = (("email", self.email_url), ("signal", self.signal_url)) 

77 return [name for name, url in offered if url]

79 def destinations(self, person: Person) -> list[str]: 

80 """Where ``person``'s notifications go: an Apprise URL per contact a channel can use.""" 

81 urls = [] 

82 if self.email_url and person.email: 

83 urls.append(with_recipient(self.email_url, person.email)) 

84 if self.signal_url and person.phone: 

85 urls.append(with_target(self.signal_url, person.phone)) 

86 return urls

88 async def send_to( 

89 self, 

90 people: Iterable[Person], 

91 event: str, 

92 message: str, 

93 values: Callable[[Person, str], dict[str, object]], 

94 ) -> int: 

95 """Send ``message`` to each person who wants ``event`` and has somewhere to get it, 

96 filled in for them and in their language by ``values(person, lang)``. Returns how many 

97 were sent.""" 

98 sent = 0 

99 for person in people: 

100 urls = self.destinations(person) 

101 if not urls or not person.wants(event): 

102 continue 

103 lang = language(person.language) 

104 title, body = render(message, lang, **values(person, lang)) 

105 try: 

106 if await self.sender.send(urls, title, body): 

107 sent += 1 

108 except Exception: # noqa: BLE001 - one failed delivery must not stop the others 

109 logger.exception("a notification could not be sent") 

110 return sent
