1"""Runtime configuration, read from the environment. 

2 

3Built on jigkit's ``AppSettings``, which supplies the data directory, the database URL, the log 

4level, ``PUBLIC_URL``, ``COOKIE_SECURE``, ``FORWARDED_ALLOW_IPS``, ``TZ``, the scheduler switch, 

5and the rules every app shares: a blank variable means unset, and a secret can be given as 

6``X`` or ``X_FILE`` but not both. 

7""" 

8 

9from functools import lru_cache 

10from pathlib import Path 

11 

12from jigkit.settings import AppSettings 

13from pydantic import AliasChoices, Field 

14 

15 

16class Settings(AppSettings): 

17 """Everything the application reads from its environment.""" 

18 

19 database_filename = "house-cost.db" 

20 

21 login_attempts_per_minute: int = 10 

22 # The largest file accepted as an attachment. 

23 max_upload_bytes: int = 20 * 1024 * 1024 

24 

25 # The daily backup: from what time of day, and how many daily snapshots are kept. 

26 backup_time: str = Field("03:30", pattern=r"^([01]\d|2[0-3]):[0-5]\d$") 

27 backups_kept: int = Field(14, ge=1) 

28 # The scheduled work runs inside the app every ten minutes; tests turn it off and call it. 

29 scheduler_tick_seconds: int = Field(600, ge=1) 

30 

31 # How notifications are sent, by Apprise URLs without a recipient: a person's email address 

32 # is added to the mail URL as its "to", their phone number to the Signal URL's path. A 

33 # channel left unset is not offered. 

34 notify_email_url: str | None = None 

35 notify_signal_url: str | None = None 

36 # The country a phone number typed without one is in: "912 345 678" is +351 912 345 678. 

37 phone_country_code: str = Field("351", pattern=r"^\d{1,3}$") 

38 

39 # Claude, for reading bills no parser knows, contracts' terms and questions about documents. 

40 # Off without a key; the key can instead be read from a file, such as a Docker secret. 

41 anthropic_api_key: str | None = None 

42 anthropic_api_key_file: Path | None = None 

43 claude_model: str = "claude-opus-5-5" 

44 

45 # The commit the image was built from, which the Dockerfile records; unset outside an image. 

46 revision: str | None = Field( 

47 None, validation_alias=AliasChoices("HOUSE_COST_REVISION", "revision") 

48 ) 

49 

50 @property 

51 def claude_key(self) -> str | None: 

52 """The Anthropic API key, from the variable or its file (jigkit reads the file).""" 

53 return self.anthropic_api_key.strip() if self.anthropic_api_key else None 

54 

55 @property 

56 def backups_dir(self) -> Path: 

57 """Where the daily snapshots of the database are kept.""" 

58 return self.data_dir / "backups" 

59 

60 @property 

61 def files_dir(self) -> Path: 

62 """Where attached files are kept, each under its SHA-256.""" 

63 return self.data_dir / "files" 

64 

65 

66@lru_cache 

67def get_settings() -> Settings: 

68 """The process-wide settings, read once.""" 

69 return Settings()
