3Exits non-zero unless the API answers and reports its database reachable. A probe that only

18 """Report whether the application is serving.""" 

19 try: 

20 with urllib.request.urlopen(URL, timeout=TIMEOUT_SECONDS) as response: # noqa: S310 

21 # urlopen raises for 4xx and 5xx, so this only catches a 2xx that is not 200 -- 

22 # a 204, say, which is not an answer to a health question either. 

23 if response.status != 200: 

24 print(f"unhealthy: HTTP {response.status}", file=sys.stderr) 

25 return 1 

26 body = json.load(response) 

27 except urllib.error.HTTPError as exc: 

28 # Closed explicitly: an HTTPError carries the response it was raised from, and letting 

29 # it be collected unclosed emits a ResourceWarning. 

30 with exc: 

31 print(f"unhealthy: HTTP {exc.code}", file=sys.stderr) 

32 return 1 

33 except (urllib.error.URLError, TimeoutError, json.JSONDecodeError) as exc: 

34 print(f"unhealthy: {exc}", file=sys.stderr) 

35 return 1 

36 

37 if body.get("database") != "connected": 

38 print(f"unhealthy: database {body.get('database')!r}", file=sys.stderr)
