9# Deterministic constraint names. SQLite cannot ALTER a constraint, so Alembic rewrites 

10# the whole table to change one — which it can only do when every constraint has a name 

11# it can reproduce. Without this, migrations that touch a constraint fail on SQLite. 

12NAMING_CONVENTION = { 

13 "ix": "ix_%(column_0_label)s", 

14 "uq": "uq_%(table_name)s_%(column_0_name)s", 

15 "ck": "ck_%(table_name)s_%(constraint_name)s", 

16 "fk": "fk_%(table_name)s_%(column_0_name)s_%(referred_table_name)s", 

17 "pk": "pk_%(table_name)s", 

18}

21class UtcDateTime(TypeDecorator): 

22 """A timestamp that is always a timezone-aware UTC ``datetime`` in Python. 

23 

24 SQLite has no timestamp type: it stores a string and hands back a naive 

25 ``datetime``, so comparing a loaded value against ``datetime.now(UTC)`` raises 

26 TypeError. Normalising to UTC on the way in and re-attaching the zone on the way 

27 out keeps every datetime in the application aware, whatever the backend. Naive 

28 input is rejected rather than assumed to be UTC, because the assumption is wrong 

29 exactly when a caller has a local time in hand. 

30 """ 

31 

32 impl = DateTime 

33 cache_ok = True 

34 

35 def process_bind_param(self, value: datetime | None, dialect: Any) -> datetime | None: 

36 if value is None: 

37 return None 

38 if value.tzinfo is None: 

39 raise ValueError(f"expected a timezone-aware datetime, got {value!r}") 

40 return value.astimezone(UTC).replace(tzinfo=None) 

41 

42 def process_result_value(self, value: datetime | None, dialect: Any) -> datetime | None: 

43 if value is None: 

44 return None 

45 return value.replace(tzinfo=UTC) 

46 

47 

48def utcnow() -> datetime: 

49 """The current time, timezone-aware, for column defaults.""" 

50 return datetime.now(UTC)
