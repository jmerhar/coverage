1"""The Tuya source: every device's report log, polled into one table per device category. 

2 

3Tuya keeps about a week of report logs, so the source polls continuously and keeps a cursor per 

4device in its state directory. The cursor is also the device's published progress 

5(``complete_until``): every report with an earlier event time has been fetched and delivered. 

6 

7Each poll fetches a device's log from ``cursor - overlap`` to now, and once a day from 

8``cursor - rescan`` instead, so reports Tuya surfaces late are still stored; re-fetched reports 

9overwrite themselves. The cursor then moves to ``poll start - lag``, never backwards, and only 

10once the batch is safe (written or spooled). A device that reported nothing still advances: an 

11empty, fully fetched window means there was nothing to report. 

12""" 

13

37class TuyaParams(BaseModel): 

38 """A Tuya source's ``params`` in the configuration file.""" 

39 

40 model_config = ConfigDict(extra="forbid") 

41 

42 #: The cloud project's data centre. 

43 region: str = "eu" 

44 #: Overrides the region's host; for tests and proxies. 

45 base_url: str | None = None 

46 access_id_file: Path 

47 access_secret_file: Path 

48 #: Tables are named this prefix plus the device category, such as ``tuya_cz`` for plugs. 

49 table_prefix: str = "tuya_" 

50 #: Only these device IDs; all devices when unset. 

51 devices: list[str] | None = None 

52 #: Integer codes stored as Tuya reports them, without the specification's scale. 

53 unscaled_codes: list[str] = Field(default_factory=lambda: ["add_ele"]) 

54 poll_minutes: float = Field(15, ge=1) 

55 #: How far behind the poll's start a device's progress stays. 

56 lag_minutes: float = Field(60, ge=0) 

57 #: How far before the cursor each poll starts fetching again. 

58 overlap_minutes: float = Field(120, ge=0) 

59 #: How far before the cursor the daily rescan starts, and how often it runs. 

60 rescan_hours: float = Field(48, ge=0) 

61 rescan_every_hours: float = Field(24, gt=0) 

62 #: How far back a device without a cursor is fetched; Tuya keeps about a week. 

63 retention_days: float = Field(7, gt=0) 

64 #: How often the device list and specifications are fetched again. 

65 discover_hours: float = Field(6, gt=0)

67 @field_validator("region") 

68 @classmethod 

69 def _known_region(cls, value: str) -> str: 

70 """Reject a data centre the client does not know.""" 

71 if value not in HOSTS: 

72 raise ValueError(f"unknown region {value!r}; one of {', '.join(sorted(HOSTS))}") 

73 return value 

74 

75 @field_validator("table_prefix") 

76 @classmethod 

77 def _table_prefix(cls, value: str) -> str: 

78 """The prefix must start a valid table name.""" 

79 if not value or not value[0].isalpha() or not value.replace("_", "").isalnum(): 

80 raise ValueError("must start with a letter and hold only letters, digits and '_'") 

81 return value

83 @model_validator(mode="after") 

84 def _overlap_covers_lag(self) -> TuyaParams: 

85 """Each poll must re-fetch at least the lag, or reports inside it could be missed.""" 

86 if self.overlap_minutes < self.lag_minutes: 

87 raise ValueError("overlap_minutes must be at least lag_minutes") 

88 return self

90 @property 

91 def url(self) -> str:

104@dataclass 

105class _State: 

106 """What survives a restart: each device's cursor and when the last rescan ran.""" 

107 

108 cursors: dict[str, int] = field(default_factory=dict) 

109 last_rescan_ms: int = 0 

110 

111 @classmethod 

112 def load(cls, path: Path) -> _State: 

113 """The saved state, or a fresh one when there is none or it cannot be read.""" 

114 try: 

115 data: dict[str, Any] = json.loads(path.read_text(encoding="utf-8")) 

116 return cls( 

117 {str(k): int(v) for k, v in dict(data.get("cursors") or {}).items()}, 

118 int(data.get("last_rescan_ms") or 0), 

119 ) 

120 except FileNotFoundError: 

121 return cls() 

122 except (OSError, ValueError, TypeError, AttributeError) as error: 

123 logger.error("cannot read %s (%s); starting without cursors", path, error) 

124 return cls() 

125 

126 def save(self, path: Path) -> None: 

127 """Write the state atomically.""" 

128 temporary = path.with_name(f".{path.name}.tmp") 

129 body = {"cursors": self.cursors, "last_rescan_ms": self.last_rescan_ms} 

130 temporary.write_text(json.dumps(body, indent=2, sort_keys=True), encoding="utf-8") 

131 os.replace(temporary, path)

134class TuyaSource: 

135 """Polls a Tuya cloud project's devices.""" 

136 

137 def __init__(self, params: TuyaParams, client: TuyaClient) -> None: 

138 self._params = params 

139 self._client = client 

140 self._unscaled = frozenset(params.unscaled_codes) 

141 self._known: list[_Known] = [] 

142 self._discovered_ms: int | None = None 

143 

144 async def run(self, context: SourceContext) -> None: 

145 """Poll for ever, one poll every ``poll_minutes``.""" 

146 path = context.state_dir / STATE_FILE 

147 state = _State.load(path) 

148 context.restore_progress([Progress(d, c) for d, c in state.cursors.items()]) 

149 while True: 

150 await self.poll(context, state, path) 

151 context.succeeded() 

152 await context.sleep(self._params.poll_minutes * 60) 

153 

154 async def poll(self, context: SourceContext, state: _State, path: Path) -> None: 

155 """Fetch every device's new reports once. 

156 

157 A device that fails is reported and skipped, so one broken device does not hold back the 

158 rest; the poll fails only when every device does. 

159 """ 

160 now = context.now_ms() 

161 if self._discovered_ms is None or now - self._discovered_ms >= ( 

162 self._params.discover_hours * HOUR_MS 

163 ): 

164 await self.discover() 

165 self._discovered_ms = now 

166 rescan = now - state.last_rescan_ms >= self._params.rescan_every_hours * HOUR_MS 

167 failures = [] 

168 for known in self._known: 

169 try: 

170 await self._poll_device(context, state, path, known, rescan) 

171 except TuyaError as error: 

172 failures.append(known.device.id) 

173 context.problem(f"device {known.device.id}: {error}") 

174 logger.error("Tuya device %s: %s", known.device.id, error) 

175 if self._known and len(failures) == len(self._known): 

176 raise TuyaError(f"every device failed ({len(failures)})") 

177 if rescan and not failures: 

178 state.last_rescan_ms = now 

179 state.save(path) 

180 

181 async def discover(self) -> None: 

182 """Fetch the device list and each device's specification. 

183 

184 A device whose specification cannot be fetched is kept without one, so it fails on its 

185 own in each poll (and is retried at the next discovery) instead of failing every device. 

186 """ 

187 wanted = set(self._params.devices) if self._params.devices else None 

188 devices = [d for d in await self._client.devices() if wanted is None or d.id in wanted] 

189 known = [] 

190 for device in devices: 

191 try: 

192 specs = parse_specification(await self._client.specification(device.id)) 

193 except TuyaError as error: 

194 logger.error("Tuya device %s: no specification: %s", device.id, error) 

195 specs = {} 

196 known.append(_Known(device, specs)) 

197 self._known = known 

198 logger.info("Tuya: %d device(s)", len(known)) 

199 

200 async def _poll_device( 

201 self, 

202 context: SourceContext, 

203 state: _State, 

204 path: Path, 

205 known: _Known, 

206 rescan: bool, 

207 ) -> None: 

208 """Fetch one device's window, deliver it with its progress, and move its cursor.""" 

209 device, params = known.device, self._params 

210 if not known.specs: 

211 raise TuyaError("its specification lists no status codes") 

212 start_poll = context.now_ms() 

213 oldest = start_poll - int(params.retention_days * DAY_MS) 

214 cursor = state.cursors.get(device.id) 

215 if cursor is None: 

216 start = oldest 

217 else: 

218 reach = params.rescan_hours * HOUR_MS if rescan else params.overlap_minutes * MINUTE_MS 

219 start = max(oldest, cursor - int(reach)) 

220 reports = await self._client.reports(device.id, sorted(known.specs), start, start_poll) 

221 built = build_points( 

222 reports, 

223 table=params.table_prefix + device.category, 

224 device_id=device.id, 

225 category=device.category, 

226 name=device.name, 

227 specs=known.specs, 

228 unscaled=self._unscaled, 

229 ) 

230 if built.collisions: 

231 context.problem(f"device {device.id}: {built.collisions} colliding report(s)") 

232 complete = start_poll - int(params.lag_minutes * MINUTE_MS) 

233 if cursor is not None: 

234 complete = max(complete, cursor) 

235 if await context.emit(built.points, [Progress(device.id, complete, device.name or None)]): 

236 state.cursors[device.id] = complete 

237 state.save(path) 

238 else: 

239 context.problem(f"device {device.id}: a batch was refused; its cursor stays put")
