1"""A minimal Modbus TCP client that can only read registers. 

2 

3It speaks function codes 3 (read holding registers) and 4 (read input registers) and nothing 

4else, so a source built on it can never change a device's settings. One request is in flight at a 

5time on one connection, kept open between reads. 

6""" 

7 

8from __future__ import annotations 

9 

10import asyncio 

11import contextlib 

12import struct 

13 

14#: Function codes this client sends: read holding registers and read input registers. 

15READ_FUNCTIONS = frozenset({3, 4}) 

16#: The most registers one read may ask for, by the Modbus specification. 

17MAX_REGISTERS = 125 

18#: Modbus exception 2: the device has no register at an address asked for. 

19ILLEGAL_ADDRESS = 2 

20 

21 

22class ModbusError(Exception): 

23 """The device answered a read with a Modbus exception."""

26 super().__init__( 

27 f"Modbus exception {code} reading {count} register(s) at {address} " 

28 f"(function {function})" 

29 ) 

30 self.code = code

33class ModbusClient: 

34 """One connection to a Modbus TCP device, opened on the first read and reopened after a 

35 failure.""" 

36 

37 def __init__(self, host: str, port: int, unit: int, timeout: float) -> None: 

38 self._host = host 

39 self._port = port 

40 self._unit = unit 

41 self._timeout = timeout 

42 self._streams: tuple[asyncio.StreamReader, asyncio.StreamWriter] | None = None 

43 self._transaction = 0

45 async def read(self, function: int, address: int, count: int) -> list[int]: 

46 """Read ``count`` registers from ``address`` as unsigned 16-bit values. 

47 

48 A Modbus exception raises :class:`ModbusError` and leaves the connection open; anything 

49 else (a timeout, a dropped connection, a malformed reply) closes it and raises, so the 

50 next read starts afresh. 

51 """ 

52 if function not in READ_FUNCTIONS: 

53 raise ValueError(f"function {function} is not a read") 

54 if not 1 <= count <= MAX_REGISTERS: 

55 raise ValueError(f"cannot read {count} registers at once") 

56 try: 

57 async with asyncio.timeout(self._timeout): 

58 return await self._exchange(function, address, count) 

59 except ModbusError: 

60 raise 

61 except BaseException: 

62 await self.close() 

63 raise

65 async def _exchange(self, function: int, address: int, count: int) -> list[int]: 

66 reader, writer = await self._connection() 

67 self._transaction = (self._transaction + 1) % 0x10000 

68 tid = self._transaction 

69 writer.write(struct.pack(">HHHBBHH", tid, 0, 6, self._unit, function, address, count)) 

70 await writer.drain() 

71 head = await reader.readexactly(7) 

72 got_tid, protocol, length, _unit = struct.unpack(">HHHB", head) 

73 body = await reader.readexactly(max(0, length - 1)) 

74 if got_tid != tid or protocol != 0 or not body: 

75 raise ConnectionError("malformed Modbus reply") 

76 if body[0] == function | 0x80: 

77 raise ModbusError(body[1] if len(body) > 1 else 0, function, address, count) 

78 if body[0] != function or len(body) < 2 or body[1] != 2 * count: 

79 raise ConnectionError("unexpected Modbus reply") 

80 return list(struct.unpack(f">{count}H", body[2 : 2 + 2 * count])) 

81

83 if self._streams is None:
