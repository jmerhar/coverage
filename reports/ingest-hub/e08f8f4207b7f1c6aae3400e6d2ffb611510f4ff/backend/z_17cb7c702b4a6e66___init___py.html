1"""Vaillant heating systems through the myVAILLANT cloud, using the myPyllant library. 

2 

3Two kinds of data, on two schedules: 

4 

5- **Energy**, every ``energy_minutes``: each device's hourly buckets for every operation mode and 

6 energy type the device reports, in Wh, into ``<prefix>energy`` with tags ``device`` (the 

7 device type, lower-cased), ``operation_mode`` and ``energy_type``, at the bucket's start. 

8 Buckets arrive late and are revised, so each poll re-reads ``rescan_hours`` before the device's 

9 cursor; the cursor, published as the device's progress, trails the poll by ``lag_hours``. 

10 A device without a cursor is read from its first data. 

11- **Live state**, every ``state_minutes``: current power per device (``<prefix>power``) and the 

12 temperatures, pressures, setpoints and states of the system, its circuits, zones and hot-water 

13 cylinders (``<prefix>system``, ``…circuit``, ``…zone``, ``…dhw``), stamped when read. 

14 

15Vaillant enforces request quotas and has frequent incidents. A refused call (HTTP 429, or 403 naming 

16the quota) or a server error, at login or later, is waited out for as long as Vaillant says or at 

17most ``pause_minutes``, rather than retried at once; bucket series are requested one at a time, 

18a moment apart. 

19"""

43HOUR = dt.timedelta(hours=1) 

44STATE_FILE = "vaillant.json" 

45#: A session this close to expiry is refreshed before the next call. 

46REFRESH_MARGIN = dt.timedelta(minutes=2) 

47#: HTTP statuses that mean "not now" rather than "never"; a 403 does too when it names the quota. 

48TRANSIENT = frozenset({429, 500, 502, 503, 504}) 

49#: How Vaillant says when a spent quota comes back: "replenished in 00:00:49". 

50REPLENISHED = re.compile(r"replenished in (\d+):(\d{2}):(\d{2})") 

51#: Seconds added to the replenishment time Vaillant names, so the next call is not early. 

52QUOTA_MARGIN = 10 

53#: The first energy poll waits this long after the first state poll, so the two bursts of 

54#: requests do not land together. 

55ENERGY_DELAY = 60.0 

56#: Seconds between one bucket series and the next, to stay under the per-minute quota. 

57SERIES_SPACING = 2.0

98def _slug(value: object) -> str: 

99 """A Vaillant enum value as a tag value: ``HEATPUMP`` becomes ``heatpump``."""

103def _transient(error: BaseException) -> bool: 

104 """Whether a failed call means "not now": a quota refusal (429, or 403 naming the quota) or 

105 a server error.""" 

106 status = _http_status(error) 

107 if status == 403: 

108 return "quota" in str(error).lower() 

109 return status in TRANSIENT

112def _pause_seconds(error: BaseException, longest: float) -> float: 

113 """How long to wait after a refusal: the replenishment time Vaillant names plus a margin, 

114 else ``longest``, and never more than ``longest``.""" 

115 match = REPLENISHED.search(str(error)) 

116 if match is None: 

117 return longest 

118 hours, minutes, seconds = (int(part) for part in match.groups()) 

119 return min(longest, hours * 3600 + minutes * 60 + seconds + QUOTA_MARGIN)

122def _http_status(error: BaseException) -> int | None: 

123 """The HTTP status of a failed call, when it was one.""" 

124 status = getattr(error, "status", None) 

125 return status if isinstance(status, int) else None

128def _numbers( 

129 record: dict[str, Any], *, skip: tuple[str, ...] = ("index",) 

130) -> dict[str, FieldValue]: 

131 """A state record's scalar values as fields: numbers as floats, flags and states as such.""" 

132 fields: dict[str, FieldValue] = {} 

133 for key, value in record.items(): 

134 # Quick-veto start and end times are moments, not measurements. 

135 if key in skip or value is None or key.endswith("_date_time"): 

136 continue 

137 if isinstance(value, bool): 

138 fields[key] = value 

139 elif isinstance(value, int | float): 

140 fields[key] = float(value) 

141 elif isinstance(value, str) and value and len(value) <= 64: 

142 fields[key] = value 

143 return fields

146class VaillantSource: 

147 """Polls one myVAILLANT account's systems.""" 

148 

149 def __init__(self, params: VaillantParams, connect: Connect, resolution: Any) -> None: 

150 self._params = params 

151 self._connect = connect 

152 self._hour = resolution 

153 self._prefix = params.table_prefix 

154 

155 async def run(self, context: SourceContext) -> None: 

156 """Log in, then poll state and energy on their schedules until something fails. 

157 

158 A quota or server error, at login or later, is waited out and the session kept or opened 

159 again, so a source that is refused does not spend more of the quota restarting. 

160 """ 

161 path = context.state_dir / STATE_FILE 

162 cursors = _load(path) 

163 context.restore_progress([Progress(d, c) for d, c in cursors.items()]) 

164 schedule = {"state": context.clock(), "energy": context.clock() + ENERGY_DELAY} 

165 while True: 

166 try: 

167 async with self._connect() as api: 

168 await self._poll_forever(context, api, cursors, path, schedule) 

169 except Exception as error: 

170 if not _transient(error): 

171 raise 

172 await self._pause(context, error) 

173 

174 async def _poll_forever( 

175 self, 

176 context: SourceContext, 

177 api: Api, 

178 cursors: dict[str, int], 

179 path: Path, 

180 schedule: dict[str, float], 

181 ) -> None: 

182 """Poll whatever is due, then wait for the next thing that will be.""" 

183 while True: 

184 now = context.clock() 

185 try: 

186 await self._refresh(api, now) 

187 if now >= schedule["state"]: 

188 await self.poll_state(context, api) 

189 schedule["state"] = now + self._params.state_minutes * 60 

190 if now >= schedule["energy"]: 

191 await self.poll_energy(context, api, cursors, path) 

192 schedule["energy"] = now + self._params.energy_minutes * 60 

193 except Exception as error: 

194 if not _transient(error): 

195 raise 

196 await self._pause(context, error) 

197 continue 

198 context.succeeded() 

199 await context.sleep(max(1.0, min(schedule.values()) - context.clock())) 

200 

201 async def _pause(self, context: SourceContext, error: BaseException) -> None: 

202 """Wait out a quota or server error: as long as Vaillant says, or ``pause_minutes``.""" 

203 seconds = _pause_seconds(error, self._params.pause_minutes * 60) 

204 message = f"Vaillant answered HTTP {_http_status(error)}; pausing {seconds:.0f} s" 

205 context.problem(message) 

206 logger.warning(message) 

207 await context.sleep(seconds) 

208 

209 async def _refresh(self, api: Api, now: float) -> None: 

210 """Renew the session when it is about to expire.""" 

211 expires = api.oauth_session_expires 

212 if expires is None: 

213 return 

214 moment = dt.datetime.fromtimestamp(now, dt.UTC) 

215 if expires.astimezone(dt.UTC) - moment <= REFRESH_MARGIN: 

216 await api.refresh_token()

218 async def poll_state(self, context: SourceContext, api: Api) -> None: 

219 """Read every system's live state and current power, stamped now.""" 

220 stamp = int(context.clock()) 

221 points: list[Point] = [] 

222 async for system in api.get_systems(include_connection_status=True, include_mpc=True): 

223 types = {getattr(d, "device_uuid", None): _slug(d.device_type) for d in system.devices} 

224 for entry in (system.mpc or {}).get("devices", []): 

225 power = entry.get("current_power") 

226 device = types.get(entry.get("device_id")) 

227 if device and isinstance(power, int | float): 

228 points.append( 

229 self._point( 

230 "power", {"device": device}, stamp, {"current_power": float(power)} 

231 ) 

232 ) 

233 state = system.state or {} 

234 if fields := _numbers(state.get("system") or {}): 

235 points.append(self._point("system", {}, stamp, fields)) 

236 for group, table in (("circuits", "circuit"), ("zones", "zone"), ("dhw", "dhw")): 

237 for record in state.get(group) or []: 

238 if fields := _numbers(record): 

239 tags = {table: str(record.get("index", 0))} 

240 points.append(self._point(table, tags, stamp, fields)) 

241 await context.emit(points) 

242 

243 async def poll_energy( 

244 self, context: SourceContext, api: Api, cursors: dict[str, int], path: Path 

245 ) -> None: 

246 """Read each device's hourly buckets from its cursor (less the rescan) up to now.""" 

247 params = self._params 

248 now = dt.datetime.fromtimestamp(context.clock(), dt.UTC) 

249 async for system in api.get_systems(): 

250 for device in system.devices: 

251 name = _slug(device.device_type) 

252 cursor = cursors.get(name) 

253 if cursor is None: 

254 first = getattr(device, "first_data", None) or now - dt.timedelta(days=7) 

255 start = first.astimezone(dt.UTC) 

256 else: 

257 start = dt.datetime.fromtimestamp(cursor / 1000, dt.UTC) - dt.timedelta( 

258 hours=params.rescan_hours 

259 ) 

260 start = start.replace(minute=0, second=0, microsecond=0) 

261 product = str(getattr(device, "product_name", "") or "") 

262 points = [] 

263 async for series in self._series(context, api, device, start, now): 

264 tags = { 

265 "device": name, 

266 "operation_mode": _slug(series.operation_mode), 

267 "energy_type": _slug(series.energy_type or series.value_type), 

268 } 

269 for bucket in series.data: 

270 if bucket.value is None: 

271 continue 

272 fields: dict[str, FieldValue] = {"wh": float(bucket.value)} 

273 if product: 

274 fields["product"] = product 

275 moment = int(bucket.start_date.astimezone(dt.UTC).timestamp()) 

276 points.append(self._point("energy", tags, moment, fields)) 

277 complete_dt = now - dt.timedelta(hours=params.lag_hours) 

278 complete = int(complete_dt.timestamp() * 1000) 

279 if cursor is not None: 

280 complete = max(complete, cursor) 

281 if await context.emit(points, [Progress(name, complete, product or None)]): 

282 cursors[name] = complete 

283 _save(path, cursors) 

284 else: 

285 context.problem(f"{name}: a batch was refused; its cursor stays put") 

286 

287 async def _series( 

288 self, context: SourceContext, api: Api, device: Any, start: dt.datetime, end: dt.datetime 

289 ) -> AsyncIterator[Any]: 

290 """A device's bucket series one request at a time, spaced out to respect the quota.""" 

291 wanted = list(getattr(device, "data", None) or []) 

292 for n, entry in enumerate(wanted or [None]): 

293 if n: 

294 await context.sleep(SERIES_SPACING) 

295 single = copy.copy(device) 

296 if entry is not None: 

297 single.data = [entry] 

298 async for series in api.get_data_by_device(single, self._hour, start, end): 

299 yield series

301 def _point( 

302 self, table: str, tags: dict[str, str], stamp: int, fields: dict[str, FieldValue] 

303 ) -> Point: 

304 """A point in one of the source's tables, at second precision.""" 

305 return Point( 

306 self._prefix + table, 

307 tuple(tags.items()), 

308 stamp, 

309 fields, 

310 precision=Precision.SECONDS, 

311 )

314def _load(path: Path) -> dict[str, int]: 

315 """The saved cursors, or none when there are none or they cannot be read.""" 

316 try: 

317 return {str(k): int(v) for k, v in json.loads(path.read_text(encoding="utf-8")).items()} 

318 except FileNotFoundError: 

319 return {} 

320 except (OSError, ValueError, TypeError, AttributeError) as error: 

321 logger.error("cannot read %s (%s); starting without cursors", path, error) 

322 return {}

325def _save(path: Path, cursors: dict[str, int]) -> None: 

326 """Write the cursors atomically.""" 

327 temporary = path.with_name(f".{path.name}.tmp") 

328 temporary.write_text(json.dumps(cursors, indent=2, sort_keys=True), encoding="utf-8") 

329 os.replace(temporary, path)

332def build(params: VaillantParams, _client: httpx2.AsyncClient, _name: str = "") -> VaillantSource: 

333 """A Vaillant source from its validated parameters, with its credentials read now.""" 

334 from myPyllant.api import MyPyllantAPI 

335 from myPyllant.enums import DeviceDataBucketResolution

337 username = read_secret(params.username_file) 

338 password = read_secret(params.password_file)
