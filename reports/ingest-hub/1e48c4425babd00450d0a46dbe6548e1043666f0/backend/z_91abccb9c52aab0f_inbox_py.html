20async def wait_or_wake(context: SourceContext, woken: asyncio.Event, seconds: float) -> None: 

21 """Wait ``seconds`` for the next scan, or until ``woken`` is set, such as by an upload.""" 

22 timer = asyncio.ensure_future(context.sleep(seconds)) 

23 wake = asyncio.ensure_future(woken.wait()) 

24 done, pending = await asyncio.wait({timer, wake}, return_when=asyncio.FIRST_COMPLETED) 

25 for task in pending: 

26 task.cancel() 

27 with contextlib.suppress(asyncio.CancelledError): 

28 await task 

29 for task in done: 

30 task.result()

33def move_aside(path: Path, folder: str) -> None: 

34 """Move a file into a folder beside it, replacing an earlier file of the same name.""" 

35 target = path.parent / folder 

36 target.mkdir(exist_ok=True) 

37 os.replace(path, target / path.name)

40def write_json(path: Path, data: Any, *, indent: int | None = None) -> None: 

41 """Write a JSON file atomically, creating its folder.""" 

42 path.parent.mkdir(parents=True, exist_ok=True) 

43 temporary = path.with_name(f".{path.name}.tmp") 

44 temporary.write_text(json.dumps(data, indent=indent), encoding="utf-8") 

45 os.replace(temporary, path)
