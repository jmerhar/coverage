1"""What a Frank Energie customer's login adds to the public prices. 

2 

3Every ``account_minutes`` the source logs in (once, and again when Frank refuses the token) and 

4writes, with ``table_prefix`` (``frank_``) before each table: 

5 

6- ``price``: the contract's own prices, tagged ``carrier`` (``electricity``, ``gas``), 

7 ``resolution`` and ``unit``: ``market_price_excl_vat``, ``market_price_incl_vat``, 

8 ``surcharge_incl_vat``, ``energy_tax_incl_vat`` and ``total_incl_vat``. These are what the 

9 customer pays, which can differ from the public prices in surcharge and resolution. 

10- ``usage``: what Frank bills per hour, tagged ``carrier`` (``electricity``, ``gas``, ``feed_in``): 

11 ``usage`` (kWh or m³) and ``costs`` (EUR). Frank has a day's figures a day or two late and can 

12 report a day before every hour is in, so a day counts as complete only once each carrier that 

13 has hours has all of them; the stream ``usage`` in ``ingest_progress`` says how far they are. 

14- ``battery``: each smart-trading battery's state, tagged ``battery``: ``soc``, ``total_result`` 

15 (EUR), ``status``, ``mode``, ``strategy`` and ``algorithm``. 

16- ``battery_day``: each day's trading, tagged ``battery``, at the day's start: ``result`` and 

17 ``cumulative_result`` (EUR), ``status``, and the day's ``epex_result``, ``frank_slim``, 

18 ``imbalance_result``, ``trading_result`` and ``total_result``. 

19- ``heating``: each smart-heating device's state, tagged ``device``: ``total_result``, 

20 ``total_savings``, ``total_bonus`` (EUR), ``operational_status``, ``steering_status`` and 

21 ``mode``. 

22- ``heating_day``: each day's smart heating, tagged ``device``: ``result``, ``savings``, 

23 ``bonus`` and ``cumulative_result`` (EUR) and ``status`` when Frank reports a session, and the 

24 day's ``period_result``, ``period_savings`` and ``period_bonus``. 

25- ``month``: the month so far, tagged ``site``: ``actual_costs`` and ``expected_costs`` (until the 

26 last meter reading), ``expected_month_costs``, ``completeness`` and ``last_reading``. 

27 

28Days are the site's, in ``timezone``. The first time, prices and usage are read from 

29``history_days`` back (not before the site's delivery start), battery and heating days from the 

30device's creation. After that each kind of data is read from its last complete day; a day is 

31read again until it is complete, so writing a day again writes the same points. 

32"""

141 # ------------------------------------------------------------------ requests 

142 async def _post(self, context: SourceContext, query: str, variables: dict, token: str | None): 

143 await context.sleep(SPACING) 

144 headers = {"Authorization": f"Bearer {token}"} if token else {} 

145 try: 

146 response = await self._client.post( 

147 self._url, 

148 json={"query": query, "variables": variables}, 

149 headers=headers, 

150 timeout=TIMEOUT, 

151 ) 

152 except httpx2.RequestError as error: 

153 raise FrankAccountError(f"cannot reach Frank: {type(error).__name__}") from error 

154 try: 

155 answer = response.json() 

156 except ValueError: 

157 answer = None 

158 if not isinstance(answer, dict): 

159 raise FrankAccountError(f"Frank answered HTTP {response.status_code} without JSON") 

160 return answer

162 async def _login(self, context: SourceContext) -> str: 

163 answer = await self._post( 

164 context, LOGIN, {"email": self._email, "password": self._password}, None 

165 ) 

166 token = ((answer.get("data") or {}).get("login") or {}).get("authToken") 

167 if not token: 

168 raise FrankAccountError(f"login refused: {_messages(answer) or 'no token'}") 

169 return str(token)

171 async def query(self, context: SourceContext, query: str, **variables: str) -> dict[str, Any]: 

172 """A query's data, logging in first, and once more when Frank refuses the token.""" 

173 for attempt in (1, 2): 

174 if self._token is None: 

175 self._token = await self._login(context) 

176 answer = await self._post(context, query, variables, self._token) 

177 messages = _messages(answer) 

178 if not messages: 

179 return answer.get("data") or {} 

180 if attempt == 1 and _is_auth(messages): 

181 self._token = None 

182 continue 

183 raise FrankAccountError(messages) 

184 raise AssertionError("unreachable") # pragma: no cover

186 # ------------------------------------------------------------------ the poll 

187 async def poll(self, context: SourceContext) -> None: 

188 """Read everything the login gives, writing each kind as it comes.""" 

189 path = context.state_dir / STATE_FILE 

190 state = _load(path) 

191 today = dt.datetime.fromtimestamp(context.clock(), self._zone).date() 

192 site = await self._site_info(context) 

193 start = max( 

194 today - dt.timedelta(days=self._history_days), 

195 _date(site.get("deliveryStartDate")) or dt.date.min, 

196 ) 

197 await self._prices(context, state, path, site["reference"], start, today) 

198 await self._usage(context, state, path, site, start, today) 

199 await self._month(context, site["reference"]) 

200 for battery in (await self.query(context, BATTERIES)).get("smartBatteries") or []: 

201 await self._battery(context, state, path, battery, today) 

202 for device in (await self.query(context, HEATERS)).get("smartHvacs") or []: 

203 await self._heater(context, state, path, device, today)

205 async def _site_info(self, context: SourceContext) -> dict[str, Any]: 

206 sites = (await self.query(context, SITES)).get("userSites") or [] 

207 for site in sites: 

208 if self._site in (None, site.get("reference")): 

209 return site 

210 raise FrankAccountError( 

211 f"site {self._site!r} is not among the account's" if self._site else "no sites"

218 def _point(self, table: str, tags: dict[str, str], stamp: int, fields: dict) -> Point: 

219 clean = {k: v for k, v in fields.items() if v is not None} 

220 return Point( 

221 self._prefix + table, 

222 tuple(sorted(tags.items())), 

223 stamp, 

224 clean, 

225 precision=Precision.SECONDS,

228 def _midnight(self, day: dt.date) -> int: 

229 return int(dt.datetime.combine(day, dt.time(), self._zone).timestamp())

231 # ------------------------------------------------------------------ prices 

232 async def _prices(self, context, state, path, site: str, start: dt.date, today: dt.date): 

233 """The contract's prices from the day after the last final one up to tomorrow.""" 

234 day = _next(state.get("prices")) or start 

235 while day <= today + dt.timedelta(days=1): 

236 data = await self.query(context, PRICES, date=day.isoformat(), site=site) 

237 prices = data.get("customerMarketPrices") or {} 

238 points = [ 

239 self._price_point(carrier, row) 

240 for carrier, key in (("electricity", "electricityPrices"), ("gas", "gasPrices")) 

241 for row in prices.get(key) or [] 

242 ] 

243 await self._emit(context, points) 

244 if day < today: 

245 state["prices"] = day.isoformat() 

246 _save(path, state) 

247 day += dt.timedelta(days=1)

249 def _price_point(self, carrier: str, row: dict[str, Any]) -> Point: 

250 spot, tax = float(row["marketPrice"]), float(row["marketPriceTax"]) 

251 tags = {"carrier": carrier, "resolution": str(row.get("resolution") or "PT60M")} 

252 if row.get("perUnit"): 

253 tags["unit"] = str(row["perUnit"]) 

254 return self._point( 

255 "price", 

256 tags, 

257 _stamp(row["from"]), 

258 { 

259 "vat_rate": float(self._vat_rate), 

260 "market_price_excl_vat": spot, 

261 "market_price_incl_vat": spot + tax, 

262 "surcharge_incl_vat": float(row["consumptionSourcingMarkupPrice"]), 

263 "energy_tax_incl_vat": float(row["energyTax"]), 

264 "total_incl_vat": float(row["allInPrice"]), 

265 }, 

266 )

269 async def _usage(self, context, state, path, site, start: dt.date, today: dt.date): 

270 """Billed usage per hour, up to the site's last meter reading.""" 

271 last = min(_date(site.get("lastMeterReadingDate")) or today, today - dt.timedelta(days=1)) 

272 complete = _date(state.get("usage")) 

273 day = _next(state.get("usage")) or start 

274 prefix_open = True 

275 while day <= last: 

276 data = await self.query(context, USAGE, date=day.isoformat(), site=site["reference"]) 

277 usage = data.get("periodUsageAndCosts") or {} 

278 points, whole = self._usage_points(usage, day) 

279 stale = (today - day).days > GIVE_UP_DAYS 

280 if not whole and stale: 

281 context.problem(f"usage for {day} is still incomplete; taking it as it is") 

282 if prefix_open and (whole or stale): 

283 complete = day 

284 else: 

285 prefix_open = False 

286 progress = [] 

287 if complete is not None and complete.isoformat() != state.get("usage"): 

288 state["usage"] = complete.isoformat() 

289 until = self._midnight(complete + dt.timedelta(days=1)) * 1000 

290 progress = [Progress("usage", until)] 

291 await self._emit(context, points, progress) 

292 _save(path, state) 

293 day += dt.timedelta(days=1)

295 def _usage_points(self, usage: dict[str, Any], day: dt.date) -> tuple[list[Point], bool]: 

296 """A day's hourly points, and whether every carrier with hours has all of them.""" 

297 hours = round((self._midnight(day + dt.timedelta(days=1)) - self._midnight(day)) / 3600) 

298 points, whole, any_hours = [], True, False 

299 for key, carrier in CARRIERS.items(): 

300 items = [i for i in (usage.get(key) or {}).get("items") or [] if i.get("from")] 

301 if not items: 

302 continue 

303 any_hours = True 

304 whole = whole and len({i["from"] for i in items}) >= hours 

305 for item in items: 

306 points.append( 

307 self._point( 

308 "usage", 

309 {"carrier": carrier}, 

310 _stamp(item["from"]), 

311 {"usage": _num(item.get("usage")), "costs": _num(item.get("costs"))}, 

312 ) 

313 ) 

314 return points, whole and any_hours

426 async def _days(self, context, state, path, key: str, device: dict, today: dt.date, read): 

427 """Read a device's days from the day after its last final one (or its creation) to 

428 today; a day is final when Frank says so or once it is a few days old.""" 

429 created = _date(str(device.get("createdAt") or "")[:10]) or today 

430 day = _next(state.get(key)) or created 

431 prefix_open = True 

432 while day <= today: 

433 points, final = await read(day) 

434 await self._emit(context, points) 

435 if prefix_open and (final or (today - day).days >= SETTLED_DAYS): 

436 state[key] = day.isoformat() 

437 _save(path, state) 

438 else: 

439 prefix_open = False 

440 day += dt.timedelta(days=1)

452def _session(period: dict[str, Any], day: dt.date) -> dict[str, Any]: 

453 """The period's session for ``day``, or an empty one.""" 

454 for session in period.get("sessions") or []: 

455 if session.get("date") == day.isoformat(): 

456 return session

484def _load(path: Path) -> dict[str, Any]: 

485 try: 

486 data = json.loads(path.read_text(encoding="utf-8")) 

487 except FileNotFoundError: 

488 return {} 

489 except (OSError, ValueError) as error: 

490 logger.error("cannot read %s (%s); reading the history again", path, error) 

491 return {} 

492 return data if isinstance(data, dict) else {}

495def _save(path: Path, state: dict[str, Any]) -> None: 

496 path.parent.mkdir(parents=True, exist_ok=True) 

497 temporary = path.with_name(f".{path.name}.tmp") 

498 temporary.write_text(json.dumps(state, indent=2, sort_keys=True), encoding="utf-8") 

499 os.replace(temporary, path)
