1"""Frank Energie: its public electricity market prices and, with a customer's login, their 

2account's data (see ``account.py``). 

3 

4Every ``poll_hours`` the source asks the public GraphQL ``marketPrices`` query for today and 

5tomorrow (by UTC date) and writes one point per price period, tagged by market and resolution, at 

6the period's start. With ``username_file`` and ``password_file`` it also reads the account every 

7``account_minutes``. A failed round is reported, and the source carries on with the next one. 

8 

9Each price poll asks for today and tomorrow (by UTC date) and writes one point per price period, 

10tagged by market and resolution, at the period's start. Frank publishes tomorrow's prices in the 

11afternoon, so until then tomorrow's query finds nothing; that is expected, not an error. 

12 

13Frank reports the spot price and its VAT separately and the surcharge and energy tax including 

14VAT. The excluded-VAT parts are derived with one fixed rate, in this order, so the stored values 

15are exactly what the same arithmetic gives anywhere: 

16 

17- ``market_price_incl_vat`` = spot + spot VAT 

18- ``surcharge_excl_vat`` = surcharge incl. VAT / (1 + rate) 

19- ``energy_tax_excl_vat`` = energy tax incl. VAT / (1 + rate) 

20- ``total_excl_vat`` = spot + surcharge excl. VAT + energy tax excl. VAT 

21- ``total_incl_vat`` = Frank's all-in price 

22"""

102class FrankSource: 

103 """Polls Frank's market prices for today and tomorrow.""" 

104 

105 def __init__( 

106 self, 

107 params: FrankParams, 

108 client: httpx2.AsyncClient, 

109 account: FrankAccount | None = None, 

110 ) -> None: 

111 self._params = params 

112 self._client = client 

113 self._account = account 

114 #: When each kind of round is next due. 

115 self._due: dict[str, float] = {} 

116 

117 async def run(self, context: SourceContext) -> None: 

118 """Poll the prices every ``poll_hours`` and the account every ``account_minutes``, for 

119 ever, reporting a failed round rather than raising it.""" 

120 due = self._due = {"prices": context.clock(), "account": context.clock()} 

121 if self._account is None: 

122 due.pop("account") 

123 every = { 

124 "prices": self._params.poll_hours * 3600, 

125 "account": self._params.account_minutes * 60, 

126 } 

127 while True: 

128 ok = True 

129 for name in list(due): 

130 if context.clock() >= due[name]: 

131 due[name] = context.clock() + every[name] 

132 ok = await self._attempt(context, name) and ok 

133 if ok: 

134 context.succeeded() 

135 await context.sleep(max(1.0, min(due.values()) - context.clock())) 

136 

137 async def _attempt(self, context: SourceContext, name: str) -> bool: 

138 """One round of prices or account data; a failure is reported, not raised.""" 

139 try: 

140 if name == "prices": 

141 await self.poll(context) 

142 else: 

143 assert self._account is not None 

144 if await self._account.poll(context): 

145 # History remains: the next round soon, rather than in account_minutes. 

146 self._due["account"] = context.clock() + BACKFILL_PAUSE 

147 except ( 

148 FrankError, 

149 FrankAccountError, 

150 httpx2.HTTPError, 

151 ValueError, 

152 KeyError, 

153 TypeError, 

154 ) as error: 

155 message = f"{name}: {str(error) or type(error).__name__}" 

156 context.problem(message) 

157 logger.warning("Frank %s", message) 

158 return False 

159 return True 

160 

161 async def poll(self, context: SourceContext) -> None: 

162 """Fetch and deliver today's and tomorrow's prices.""" 

163 today = datetime.fromtimestamp(context.clock(), UTC).date() 

164 for day in (today, today + timedelta(days=1)): 

165 points = await self.prices(day) 

166 if not points: 

167 logger.info("Frank: no prices for %s yet", day.isoformat()) 

168 continue 

169 await context.emit(points) 

170 

171 def resolution(self, day: date) -> str: 

172 """The resolution a day is asked for at.""" 

173 return "PT15M" if day >= self._params.quarter_hours_from else "PT60M" 

174 

175 async def prices(self, day: date) -> list[Point]: 

176 """The points for one day, or none when Frank has not published it.""" 

177 resolution = self.resolution(day) 

178 body = {"query": QUERY, "variables": {"date": day.isoformat(), "resolution": resolution}} 

179 try: 

180 response = await self._client.post(self._params.url, json=body, timeout=TIMEOUT_SECONDS) 

181 except httpx2.RequestError as error: 

182 raise FrankError(f"cannot reach Frank: {type(error).__name__}") from error 

183 try: 

184 answer = response.json() 

185 except ValueError: 

186 answer = None 

187 if not isinstance(answer, dict): 

188 raise FrankError(f"Frank answered HTTP {response.status_code} without JSON") 

189 errors = answer.get("errors") or [] 

190 if errors: 

191 messages = "; ".join(str(e.get("message", e)) for e in errors) 

192 if all(NOT_PUBLISHED in str(e.get("message", "")) for e in errors): 

193 return [] 

194 raise FrankError(f"Frank refused the query for {day.isoformat()}: {messages}") 

195 rows = ((answer.get("data") or {}).get("marketPrices") or {}).get("electricityPrices") 

196 return [self.point(row, resolution) for row in rows or []] 

197 

198 def point(self, row: dict[str, Any], resolution: str) -> Point: 

199 """One price period as a point.""" 

200 rate = self._params.vat_rate 

201 spot = float(row["marketPrice"]) 

202 spot_tax = float(row["marketPriceTax"]) 

203 surcharge = float(row["sourcingMarkupPrice"]) 

204 energy_tax = float(row["energyTaxPrice"]) 

205 surcharge_excl = surcharge / (1 + rate) 

206 energy_tax_excl = energy_tax / (1 + rate) 

207 start = datetime.fromisoformat(str(row["from"]).replace("Z", "+00:00")) 

208 return Point( 

209 self._params.table, 

210 (("market", self._params.market), ("resolution", row.get("resolution") or resolution)), 

211 int(start.timestamp()), 

212 { 

213 "vat_rate": float(rate), 

214 "market_price_excl_vat": spot, 

215 "market_price_incl_vat": spot + spot_tax, 

216 "surcharge_excl_vat": surcharge_excl, 

217 "surcharge_incl_vat": surcharge, 

218 "energy_tax_excl_vat": energy_tax_excl, 

219 "energy_tax_incl_vat": energy_tax, 

220 "total_excl_vat": spot + surcharge_excl + energy_tax_excl, 

221 "total_incl_vat": float(row["allInPrice"]), 

222 }, 

223 precision=Precision.SECONDS, 

224 ) 

225

227def build(params: FrankParams, client: httpx2.AsyncClient, _name: str = "") -> FrankSource: 

228 """A Frank source from its validated parameters, with the login read now when it has one.""" 

229 account = None 

230 if params.username_file is not None and params.password_file is not None: 

231 account = FrankAccount( 

232 client, 

233 params.url, 

234 read_secret(params.username_file), 

235 read_secret(params.password_file), 

236 prefix=params.table_prefix, 

237 timezone=params.timezone, 

238 history_days=params.history_days, 

239 vat_rate=params.vat_rate, 

240 site=params.site, 

241 quarter_hours_from=params.quarter_hours_from, 

242 ) 

243 return FrankSource(params, client, account)
