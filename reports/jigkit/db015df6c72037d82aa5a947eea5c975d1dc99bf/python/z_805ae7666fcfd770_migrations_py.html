1"""Checks an application runs against its own Alembic migrations. 

2 

3Every check runs ``alembic`` in a subprocess from the application's directory, exactly as a 

4container's entrypoint does, so the application's real env.py and settings are what is tested:: 

5 

6 from jigkit.testing.migrations import MigrationProject 

7 

8 PROJECT = MigrationProject(Path(__file__).parents[1], database_filename="myapp.db") 

9 

10 def test_models_match_migrations(data_dir): 

11 PROJECT.assert_models_match(Base.metadata, data_dir) 

12 

13 def test_one_head(): 

14 PROJECT.assert_single_head() 

15 

16 def test_rebuilding_parents_keeps_children(data_dir, tmp_path): 

17 PROJECT.assert_parent_rebuilds_keep_children(Base.metadata, data_dir, tmp_path) 

18 

19The application's tests build their schema with ``create_all``, so without these nothing would 

20notice a model change that never got a migration, or a migration that empties a table. 

21"""

77@dataclass(frozen=True) 

78class MigrationProject: 

79 """An application directory holding ``alembic.ini``, and how to point it at a database. 

80 

81 ``env`` is added to the subprocess environment, for settings the application insists on 

82 (a secret key, say). The data directory reaches the application as ``data_dir_variable``. 

83 """ 

84 

85 directory: Path 

86 database_filename: str = "app.db" 

87 env: Mapping[str, str] = field(default_factory=dict) 

88 data_dir_variable: str = "DATA_DIR" 

89 ini_name: str = "alembic.ini" 

90 

91 def config(self) -> Config: 

92 """The application's Alembic configuration, with paths made absolute.""" 

93 config = Config(str(self.directory / self.ini_name)) 

94 location = config.get_main_option("script_location") or "alembic" 

95 config.set_main_option("script_location", str(self.directory / location)) 

96 return config

98 def sync_url(self, data_dir: Path) -> str: 

99 """A synchronous URL for the SQLite database the application keeps in ``data_dir``.""" 

100 return f"sqlite:///{data_dir / self.database_filename}"

102 def alembic(self, data_dir: Path, *args: str, directory: Path | None = None) -> None: 

103 """Run ``alembic *args`` against the database in ``data_dir``; raise if it fails. 

104 

105 ``directory`` runs it from a copy of the project instead of the project itself. 

106 """ 

107 cwd = directory or self.directory 

108 environment = {k: v for k, v in os.environ.items() if k in PASSED_THROUGH} 

109 # Coverage measures subprocesses through variables of its own; keep them so the code 

110 # the migrations run is measured too. 

111 environment |= {k: v for k, v in os.environ.items() if k.startswith("COVERAGE")} 

112 environment |= {"PYTHONPATH": os.pathsep.join(self._python_path())} 

113 environment |= dict(self.env) 

114 environment[self.data_dir_variable] = str(data_dir) 

115 result = subprocess.run( 

116 [sys.executable, "-m", "alembic", "-c", self.ini_name, *args], 

117 cwd=cwd, 

118 env=environment, 

119 capture_output=True, 

120 text=True, 

121 check=False, 

122 ) 

123 if result.returncode != 0: 

124 raise AssertionError( 

125 f"alembic {' '.join(args)} failed with status {result.returncode}:\n" 

126 f"{result.stdout}{result.stderr}" 

127 )

129 def _python_path(self) -> list[str]: 

130 """The project's ``prepend_sys_path`` entries made absolute, then the current path. 

131 

132 Absolute, so the project's own modules still import when Alembic runs from a copy. 

133 """ 

134 config = Config(str(self.directory / self.ini_name)) 

135 entries = (config.get_main_option("prepend_sys_path") or "").split(os.pathsep) 

136 prepended = [str((self.directory / entry).resolve()) for entry in entries if entry] 

137 return [*prepended, *sys.path]

139 def assert_single_head(self) -> None: 

140 """Fail unless the migrations have exactly one head. 

141 

142 Two heads mean two migrations claim the same parent, and ``alembic upgrade head`` fails 

143 on a deployment rather than in CI. 

144 """ 

145 heads = ScriptDirectory.from_config(self.config()).get_heads() 

146 assert len(heads) == 1, f"expected one migration head, found {len(heads)}: {heads}"

148 def assert_models_match(self, metadata: MetaData, data_dir: Path) -> None: 

149 """Fail unless upgrading an empty database to head yields exactly ``metadata``'s schema.""" 

150 self.alembic(data_dir, "upgrade", "head") 

151 engine = create_engine(self.sync_url(data_dir)) 

152 try: 

153 with engine.connect() as connection: 

154 context = MigrationContext.configure( 

155 connection, opts={"compare_type": True, "render_as_batch": True} 

156 ) 

157 differences = compare_metadata(context, metadata) 

158 finally: 

159 engine.dispose() 

160 assert differences == [], ( 

161 f"the models and the migrations disagree; generate a migration for: {differences}" 

162 )

164 def assert_parent_rebuilds_keep_children( 

165 self, 

166 metadata: MetaData, 

167 data_dir: Path, 

168 scratch: Path, 

169 overrides: Mapping[str, Mapping[str, Any]] | None = None, 

170 extra_rows: Mapping[str, Sequence[Mapping[str, Any]]] | None = None, 

171 ) -> None: 

172 """Fail if rebuilding a table that others point at loses or orphans their rows. 

173 

174 SQLite's batch mode rebuilds a table by copying it, dropping the original and renaming 

175 the copy; with foreign keys on, the drop runs every ON DELETE rule pointing at it. This 

176 upgrades a database to head, puts one synthetic row in every table, then runs a copy of 

177 the migrations with one more that rebuilds every referenced table, and checks every row 

178 is still there. 

179 

180 Rows are invented from column types. ``overrides`` supplies values for columns whose 

181 type or constraints the generator cannot satisfy: ``{"table": {"column": value}}``. 

182 ``extra_rows`` adds further rows to a table after its synthetic one, each a synthetic row 

183 with the given values on top, for a constraint one row per table cannot meet -- two 

184 different households for a row that must name two, say:: 

185 

186 extra_rows={"household": [{"id": 2}]}, 

187 overrides={"settlement": {"to_id": 2}}, 

188 

189 ``scratch`` is an empty directory for the copy of the migrations. 

190 """ 

191 parents = sorted( 

192 {fk.column.table.name for table in metadata.sorted_tables for fk in table.foreign_keys} 

193 ) 

194 assert parents, "no table is referenced by a foreign key; there is nothing to check" 

195 

196 self.alembic(data_dir, "upgrade", "head") 

197 engine = create_engine(self.sync_url(data_dir)) 

198 try: 

199 with engine.begin() as connection: 

200 for table in metadata.sorted_tables: 

201 row = synthetic_row(table, overrides) 

202 connection.execute(table.insert().values(**row)) 

203 for extra in (extra_rows or {}).get(table.name, ()): 

204 connection.execute(table.insert().values(**(row | dict(extra)))) 

205 before = _row_counts(engine, metadata) 

206 

207 copy = self._copy_with_rebuild(scratch, parents) 

208 self.alembic(data_dir, "upgrade", "head", directory=copy) 

209 

210 after = _row_counts(engine, metadata) 

211 finally: 

212 engine.dispose() 

213 lost = {name: (before[name], after[name]) for name in before if after[name] != before[name]} 

214 assert not lost, ( 

215 f"rebuilding {', '.join(parents)} changed row counts (before, after): {lost}" 

216 )

218 def _copy_with_rebuild(self, scratch: Path, parents: list[str]) -> Path: 

219 """Copy the project's Alembic files into ``scratch`` and add the rebuilding migration.""" 

220 config = self.config() 

221 source = Path(config.get_main_option("script_location") or "") 

222 head = ScriptDirectory.from_config(config).get_current_head() 

223 shutil.copy(self.directory / self.ini_name, scratch / self.ini_name) 

224 relative = source.relative_to(self.directory) 

225 shutil.copytree(source, scratch / relative) 

226 versions = scratch / relative / "versions" 

227 (versions / f"{REBUILD_REVISION}.py").write_text( 

228 "from alembic import op\n\n" 

229 f"revision = {REBUILD_REVISION!r}\n" 

230 f"down_revision = {head!r}\n" 

231 "branch_labels = None\n" 

232 "depends_on = None\n\n\n" 

233 "def upgrade():\n" 

234 f" for name in {parents!r}:\n" 

235 " with op.batch_alter_table(name, recreate='always'):\n" 

236 " pass\n\n\n" 

237 "def downgrade():\n" 

238 " pass\n", 

239 encoding="utf-8", 

240 ) 

241 return scratch

244def _row_counts(engine: Any, metadata: MetaData) -> dict[str, int]: 

245 """The number of rows in each of ``metadata``'s tables.""" 

246 with engine.connect() as connection: 

247 return { 

248 table.name: connection.execute(select(func.count()).select_from(table)).scalar_one() 

249 for table in metadata.sorted_tables 

250 }

253def synthetic_row( 

254 table: Table, overrides: Mapping[str, Mapping[str, Any]] | None = None 

255) -> dict[str, Any]: 

256 """Values for one row of ``table`` that point every foreign key at the referenced table's row. 

257 

258 Every table gets a row with integer primary keys of 1, so a foreign key column takes the 

259 value of the column it references in the row generated for that table. Nullable columns with 

260 no foreign key and columns with a server default are left out. 

261 """ 

262 explicit = dict((overrides or {}).get(table.name, {})) 

263 row: dict[str, Any] = {} 

264 for column in table.columns: 

265 if column.name in explicit: 

266 row[column.name] = explicit[column.name] 

267 elif column.foreign_keys: 

268 # A column with several foreign keys points at the same value in each referenced row, 

269 # since every generated row has the same keys; the first one decides it. 

270 fk = sorted(column.foreign_keys, key=lambda key: key.target_fullname)[0] 

271 row[column.name] = _value_for(fk.column, table.name, overrides) 

272 elif column.primary_key or not (column.nullable or column.server_default is not None): 

273 row[column.name] = _value_for(column, table.name, overrides) 

274 return row

277def _value_for( 

278 column: Column[Any], table_name: str, overrides: Mapping[str, Mapping[str, Any]] | None 

279) -> Any: 

280 """The value the generated row of ``column``'s table holds in it.""" 

281 owner = column.table.name 

282 explicit = (overrides or {}).get(owner, {}) 

283 if column.name in explicit: 

284 return explicit[column.name] 

285 column_type = column.type 

286 # Checked before the generic table below: Enum is a String, and UtcDateTime wraps DateTime. 

287 if isinstance(column_type, Enum): 

288 return column_type.enums[0] 

289 if isinstance(column_type, UtcDateTime): 

290 return datetime(2026, 1, 1, tzinfo=UTC) 

291 simple: list[tuple[type, Any]] = [ 

292 (Boolean, True), 

293 (Integer, 1), 

294 (String, "x"), 

295 (DateTime, datetime(2026, 1, 1)), 

296 (Date, date(2026, 1, 1)), 

297 (Time, time(0, 0)), 

298 (Interval, timedelta(0)), 

299 # Float subclasses Numeric, so it comes first. 

300 (Float, 1.0), 

301 (Numeric, Decimal(1)), 

302 (LargeBinary, b"\x00"), 

303 (JSON, {}), 

304 (Uuid, uuid.UUID(int=1)), 

305 ] 

306 for kind, value in simple: 

307 if isinstance(column_type, kind): 

308 return value 

309 raise UnsupportedColumnError( 

310 f"no synthetic value for {owner}.{column.name} ({column_type!r}) while filling " 

311 f"{table_name}; pass one in overrides={{{owner!r}: {{{column.name!r}: ...}}}}" 

312 )
