1"""Password hashes: Argon2id at the library's defaults, and a length floor.""" 

2 

3from __future__ import annotations 

4 

5import secrets 

6 

7from argon2 import PasswordHasher 

8from argon2.exceptions import Argon2Error, InvalidHashError 

9 

10#: The shortest password :func:`hash_password` accepts unless told otherwise. 

11MIN_PASSWORD_LENGTH = 10 

12 

13# The library's defaults are memory-hard, and move with the library rather than with a number 

14# pinned here that nobody revisits. 

15_hasher = PasswordHasher() 

16 

17 

18class PasswordTooShort(ValueError): 

19 """A new password is shorter than the minimum.""" 

20 

21 

22def hash_password(password: str, *, min_length: int = MIN_PASSWORD_LENGTH) -> str: 

23 """An Argon2id hash of ``password``, which must be at least ``min_length`` characters. 

24 

25 A length floor is the one rule enforced in code: an application reachable from the internet 

26 has a short password as its whole attack surface. 

27 """ 

28 if len(password) < min_length: 

29 raise PasswordTooShort(f"a password needs at least {min_length} characters") 

30 return _hasher.hash(password) 

31 

32 

33def verify_password(password_hash: str, password: str) -> bool: 

34 """Whether ``password`` matches the hash; a malformed hash simply does not match.""" 

35 try: 

36 return _hasher.verify(password_hash, password) 

37 except Argon2Error, InvalidHashError: 

38 return False 

39 

40 

41#: Verified against when a login names nobody, so an unknown login takes as long to refuse as a 

42#: wrong password and timing does not reveal which logins exist. 

43_DUMMY_HASH = _hasher.hash(secrets.token_urlsafe(16)) 

44 

45 

46def burn_password_check(password: str) -> None: 

47 """Spend the time a real password check takes, for a login that names nobody.""" 

48 verify_password(_DUMMY_HASH, password)
