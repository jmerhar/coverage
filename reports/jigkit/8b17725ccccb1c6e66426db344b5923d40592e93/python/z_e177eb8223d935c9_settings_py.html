1"""Settings read from the environment, with the behaviour every application needs. 

2 

3An application subclasses :class:`AppSettings`, adds its own fields, and gets: 

4 

5- **Blank means unset.** Compose writes ``VAR: ${VAR:-}`` for an optional setting, so a variable 

6 nobody set arrives as an empty string. A blank value is dropped before validation, so the 

7 field takes its default: an empty ``STATIC_DIR`` stays ``None`` instead of becoming 

8 ``Path(".")``, and an empty ``COOKIE_SECURE`` stays "follow the scheme" instead of failing to 

9 parse. 

10- **Secrets from files.** A field ``x`` with a sibling ``x_file`` can be given either directly 

11 or as a path to a file holding it, such as a mounted Docker secret. Setting both is an error, 

12 as is a file that is empty or unreadable: either way the application would otherwise run with 

13 a value nobody intended. 

14- **A known time zone.** ``TZ`` is validated at startup, so a typo stops the application instead 

15 of surfacing later as a scheduler that cannot compute the next run. 

16""" 

17 

18from __future__ import annotations 

19 

20from pathlib import Path 

21from typing import Any, ClassVar, Self 

22from zoneinfo import ZoneInfo, ZoneInfoNotFoundError 

23 

24from pydantic import AliasChoices, Field, TypeAdapter, field_validator, model_validator 

25from pydantic_settings import BaseSettings, SettingsConfigDict

27#: The suffix that marks a field as the file form of another. 

28FILE_SUFFIX = "_file" 

29 

30 

31class AppSettings(BaseSettings): 

32 """The settings every application shares. 

33 

34 Subclass it and add fields. Set ``database_filename`` on the subclass to name the SQLite file 

35 the default database URL points at. 

36 """ 

37 

38 model_config = SettingsConfigDict(env_file=".env", extra="ignore", populate_by_name=True) 

39 

40 #: The SQLite file inside ``data_dir`` that :attr:`resolved_database_url` defaults to. 

41 database_filename: ClassVar[str] = "app.db" 

42 

43 #: Where all mutable state lives: the database, generated keys, logs, uploads. One directory 

44 #: means a deployment has exactly one thing to persist and back up. 

45 data_dir: Path = Path("data") 

46 #: Where the built single-page app is served from. Unset, the API runs on its own, which is 

47 #: how a development stack works (the frontend's dev server serves the app and proxies the API). 

48 static_dir: Path | None = None 

49 #: Overrides the default SQLite database in ``data_dir``. 

50 database_url: str | None = None 

51 

52 log_level: str = "INFO" 

53 

54 #: The address people open the application at, behind its reverse proxy. 

55 public_url: str | None = None 

56 #: Forces the session cookie's Secure flag on or off. Unset, it follows the request's scheme. 

57 cookie_secure: bool | None = None 

58 #: The proxies whose X-Forwarded-For and X-Forwarded-Proto are believed, as uvicorn's 

59 #: ``forwarded_allow_ips`` takes them. Never ``*``: uvicorn then believes the leftmost 

60 #: address, which the client wrote itself. 

61 forwarded_allow_ips: str = "127.0.0.1" 

62 

63 #: The application's time zone, read from ``TZ``, which the container's clock follows too. 

64 timezone: str = Field("UTC", validation_alias=AliasChoices("TZ", "timezone")) 

65 

66 #: The commit the image was built from; unset outside an image. 

67 revision: str | None = None 

68 

69 #: Whether in-process scheduled work runs. Tests turn it off and drive the work themselves. 

70 scheduler_enabled: bool = True 

71 scheduler_tick_seconds: int = Field(60, ge=1)

73 @model_validator(mode="before") 

74 @classmethod 

75 def _blank_means_unset(cls, data: Any) -> Any: 

76 """Drop blank string values, so their fields take their defaults.""" 

77 if not isinstance(data, dict): 

78 return data 

79 return { 

80 key: value 

81 for key, value in data.items() 

82 if not (isinstance(value, str) and not value.strip()) 

83 } 

84 

85 @field_validator("timezone") 

86 @classmethod 

87 def _known_zone(cls, value: str) -> str: 

88 """Reject a time zone the standard library does not know.""" 

89 try: 

90 ZoneInfo(value) 

91 except (ZoneInfoNotFoundError, ValueError) as error: 

92 raise ValueError(f"unknown time zone {value!r}") from error 

93 return value 

94 

95 @model_validator(mode="after") 

96 def _read_secret_files(self) -> Self: 

97 """Fill each field whose ``*_file`` companion names a file, from that file. 

98 

99 The file's stripped text is validated as the target field's type, so a number or a flag 

100 can come from a file as well as a string. The value and its file are exclusive, and the 

101 file must hold something: an unreadable or empty file is an error rather than a silent 

102 fall back to the default. 

103 """ 

104 fields = type(self).model_fields 

105 for name in fields: 

106 if not name.endswith(FILE_SUFFIX): 

107 continue 

108 target = name.removesuffix(FILE_SUFFIX) 

109 path = getattr(self, name) 

110 if target not in fields or path is None: 

111 continue 

112 env_target = target.upper() 

113 if target in self.model_fields_set: 

114 raise ValueError( 

115 f"{env_target} and {env_target}{FILE_SUFFIX.upper()} are exclusive; " 

116 "set one of them" 

117 ) 

118 try: 

119 value = Path(path).read_text(encoding="utf-8").strip() 

120 except OSError as error: 

121 raise ValueError( 

122 f"{env_target}{FILE_SUFFIX.upper()}: cannot read {path}: {error.strerror}" 

123 ) from error 

124 if not value: 

125 raise ValueError(f"{env_target}{FILE_SUFFIX.upper()}: {path} is empty") 

126 setattr(self, target, TypeAdapter(fields[target].annotation).validate_python(value)) 

127 return self 

128 

129 @property 

130 def resolved_database_url(self) -> str: 

131 """The database URL, defaulting to a SQLite file inside the data directory.""" 

132 if self.database_url: 

133 return self.database_url 

134 return f"sqlite+aiosqlite:///{self.data_dir / self.database_filename}" 

135 

136 @property 

137 def zone(self) -> ZoneInfo: 

138 """The application's time zone.""" 

139 return ZoneInfo(self.timezone)
