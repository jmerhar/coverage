1"""A fail-closed access policy: every route needs a credential unless it is declared open. 

2 

3An application lists the routes reachable without one, as ``(method, full path template)`` 

4pairs, and mounts the policy once as a global dependency:: 

5 

6 POLICY = AccessPolicy( 

7 {("GET", "/api/health"), ("POST", "/api/auth/login")}, 

8 allow=lambda request: is_logged_in(request), 

9 prefix="/api", 

10 ) 

11 app.include_router(api, prefix="/api", dependencies=[Depends(POLICY)]) 

12 

13A route added without touching the list needs the credential; the mistake that remains possible 

14is a page that breaks for logged-out visitors, not a write left open to everyone, as a 

15forgotten per-route dependency would leave it. :func:`route_templates` and 

16:meth:`AccessPolicy.unknown` let a test hold the list to the application's real route table. 

17 

18The policy keys off the route the router matched rather than the requested URL, so a 

19declaration cannot cover a neighbouring path. 

20""" 

21 

22from __future__ import annotations 

23 

24import inspect 

25from collections.abc import Awaitable, Callable, Iterable 

26from typing import Any 

27 

28from fastapi import FastAPI, HTTPException, Request, status 

29 

30#: A route as ``(method, full path template)``, such as ``("GET", "/api/items/{item_id}")``. 

31Route = tuple[str, str] 

32 

33 

34def route_templates(app: FastAPI) -> set[Route]: 

35 """Every ``(method, full path template)`` the application serves, from its OpenAPI schema. 

36 

37 The schema reports full templates however the routers were nested, which the route objects 

38 themselves do not across FastAPI versions. Routes left out of the schema 

39 (``include_in_schema=False``) are left out here too. 

40 """ 

41 return { 

42 (method.upper(), path) 

43 for path, operations in app.openapi()["paths"].items() 

44 for method in operations 

45 } 

46 

47 

48class AccessPolicy: 

49 """A FastAPI dependency refusing every request to an undeclared route without a credential. 

50 

51 ``open_routes`` are reachable by anyone; any other route needs ``allow(request)``, which 

52 may be async, to return true, or the request is refused with ``status_code`` and 

53 ``detail``. ``prefix`` is where the guarded routers are mounted: a matched route reports its 

54 path relative to its mount point in some FastAPI versions and in full in others, so both 

55 readings are checked. 

56 """ 

57 

58 def __init__( 

59 self, 

60 open_routes: Iterable[Route], 

61 *, 

62 allow: Callable[[Request], bool | Awaitable[bool]], 

63 prefix: str = "", 

64 status_code: int = status.HTTP_403_FORBIDDEN, 

65 detail: str = "forbidden", 

66 ) -> None: 

67 self.open_routes = frozenset((method.upper(), path) for method, path in open_routes) 

68 self._allow = allow 

69 self._prefix = prefix.rstrip("/") 

70 self._status_code = status_code 

71 self._detail = detail 

72 

73 def is_open(self, method: str, route_path: str) -> bool: 

74 """Whether a matched route, by its own reported path, is reachable without a credential.""" 

75 method = method.upper() 

76 return any( 

77 (method, template) in self.open_routes 

78 for template in (route_path, self._prefix + route_path) 

79 ) 

80 

81 def unknown(self, app: FastAPI) -> set[Route]: 

82 """Open routes the application does not serve: a typo, or a route since removed.""" 

83 return set(self.open_routes - route_templates(app)) 

84 

85 async def __call__(self, request: Request) -> None: 

86 """Let the request through if its route is open or the credential checks out. 

87 

88 A request that matched no route has no declaration, so it needs the credential; in 

89 practice an unmatched path is answered with a 404 before dependencies run. 

90 """ 

91 route: Any = request.scope.get("route") 

92 route_path = getattr(route, "path", None) 

93 if route_path is not None and self.is_open(request.method, route_path): 

94 return 

95 allowed = self._allow(request) 

96 if inspect.isawaitable(allowed): 

97 allowed = await allowed 

98 if not allowed: 

99 raise HTTPException(self._status_code, self._detail)
