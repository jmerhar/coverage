31def render_item(obj_type: str, obj: object, autogen_context: Any) -> str | bool: # noqa: ARG001 

32 """Render :class:`~jigkit.models.UtcDateTime` as the plain ``DateTime`` it is built on. 

33 

34 A migration has to keep running years later, and one that imports an application class stops 

35 working when that class moves. At the DDL level the type is exactly ``sa.DateTime()``. 

36 

37 Everything else returns False, Alembic's "use the default rendering"; None would render the 

38 item as an empty string. 

39 """ 

40 if obj_type == "type" and isinstance(obj, UtcDateTime): 

41 return "sa.DateTime()" 

42 return False

97def _without_foreign_keys(dbapi_connection: Any, _record: object) -> None: 

98 """Turn SQLite's foreign keys off for the migrations. 

99 

100 Batch mode rebuilds a table by copying it, dropping the original and renaming the copy. With 

101 foreign keys on, dropping the original runs every ON DELETE rule that points at it, so 

102 rebuilding a parent table would delete or orphan every child row. The check after the 

103 migrations stands in for the enforcement left off during them. 

104 """ 

105 cursor = dbapi_connection.cursor() 

106 try: 

107 cursor.execute("PRAGMA foreign_keys=OFF") 

108 finally: 

109 cursor.close()

112def _ensure_sqlite_directory(url: str) -> None: 

113 """Create the directory a SQLite file lives in, which SQLite will not do itself.""" 

114 database = make_url(url).database 

115 if database and database != ":memory:" and not database.startswith("file:"): 

116 Path(database).parent.mkdir(parents=True, exist_ok=True)
