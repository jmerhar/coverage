37#: A room is worth airing when it is at least this humid... 

38AIRING_HUMIDITY_PCT = 65.0 

39#: ...and airing would bring it at least this many points lower. Airing is advised against only 

40#: when it would do harm: leave a room at least AIRING_HUMIDITY_PCT humid and this many points 

41#: damper than it is. Humidity sensors are good to a few points, and the humidity after airing 

42#: adds the outdoor sensor's error to the room's, so a smaller difference says nothing. The page 

43#: applies the same to the house's median room, and treats a smaller change either way as none. 

44#: The Good time to air and Damp room alert rules in grafana/alerts.json use the same numbers. 

45DAMPER_BY_PCT = 5.0

82def compass(degrees: float) -> str: 

83 """The 16-point compass direction a wind blowing from `degrees` comes from.""" 

84 return COMPASS[round(degrees % 360 / 22.5) % 16]

87def beaufort(speed_kmh: float) -> int: 

88 """The Beaufort force of a wind speed.""" 

89 return bisect_right(BEAUFORT_FROM_MS, speed_kmh / 3.6)

92def wind_chill(temp_c: float, speed_kmh: float) -> float: 

93 """The North American wind chill index (2001), for 10 m wind speeds.""" 

94 power = speed_kmh**0.16 

95 return 13.12 + 0.6215 * temp_c - 11.37 * power + 0.3965 * temp_c * power

98def heat_index(temp_c: float, humidity_pct: float) -> float: 

99 """The US National Weather Service heat index: Rothfusz's regression and its adjustments.""" 

100 t = temp_c * 9 / 5 + 32 

101 rh = humidity_pct 

102 simple = 0.5 * (t + 61.0 + (t - 68.0) * 1.2 + rh * 0.094) 

103 if (simple + t) / 2 < 80: 

104 return (simple - 32) * 5 / 9 

105 index = ( 

106 -42.379 

107 + 2.04901523 * t 

108 + 10.14333127 * rh 

109 - 0.22475541 * t * rh 

110 - 0.00683783 * t * t 

111 - 0.05481717 * rh * rh 

112 + 0.00122874 * t * t * rh 

113 + 0.00085282 * t * rh * rh 

114 - 0.00000199 * t * t * rh * rh 

115 ) 

116 if rh < 13 and 80 <= t <= 112: 

117 index -= (13 - rh) / 4 * math.sqrt((17 - abs(t - 95)) / 17) 

118 elif rh > 85 and 80 <= t <= 87: 

119 index += (rh - 85) / 10 * (87 - t) / 5 

120 return (index - 32) * 5 / 9

123def feels_like(temp_c: float, humidity_pct: float | None, speed_kmh: float | None) -> float: 

124 """What the air feels like, the way the US National Weather Service reports it. 

125 

126 Wind chill when it is 10 °C or colder with wind of at least 4.8 km/h; the heat index when it 

127 is 26.7 °C (80 °F) or warmer; otherwise the air temperature itself. 

128 """ 

129 if temp_c <= 10 and speed_kmh is not None and speed_kmh >= 4.8: 

130 return wind_chill(temp_c, speed_kmh) 

131 if temp_c >= 26.7 and humidity_pct is not None: 

132 return max(temp_c, heat_index(temp_c, humidity_pct)) 

133 return temp_c

136class Tendency(Enum): 

137 """A three-hour pressure change, in the Met Office's terms.""" 

138 

139 STEADY = "steady" 

140 RISING_SLOWLY = "rising_slowly" 

141 RISING = "rising" 

142 RISING_QUICKLY = "rising_quickly" 

143 RISING_VERY_RAPIDLY = "rising_very_rapidly" 

144 FALLING_SLOWLY = "falling_slowly" 

145 FALLING = "falling" 

146 FALLING_QUICKLY = "falling_quickly" 

147 FALLING_VERY_RAPIDLY = "falling_very_rapidly"

150def tendency(change_hpa: float) -> Tendency: 

151 """The tendency of a three-hour pressure change.""" 

152 size = abs(change_hpa) 

153 if size < 0.1: 

154 return Tendency.STEADY 

155 rising = change_hpa > 0 

156 if size <= 1.5: 

157 return Tendency.RISING_SLOWLY if rising else Tendency.FALLING_SLOWLY 

158 if size <= 3.5: 

159 return Tendency.RISING if rising else Tendency.FALLING 

160 if size <= 6.0: 

161 return Tendency.RISING_QUICKLY if rising else Tendency.FALLING_QUICKLY 

162 return Tendency.RISING_VERY_RAPIDLY if rising else Tendency.FALLING_VERY_RAPIDLY

165class Airing(Enum): 

166 """Whether opening a room's windows would make it drier.""" 

167 

168 #: Damp, and airing would bring it DAMPER_BY_PCT points or more lower. 

169 OPEN = "open" 

170 #: Airing would leave it damp, and DAMPER_BY_PCT points or more damper than it is. 

171 KEEP_CLOSED = "keep_closed" 

172 #: Neither: dry enough already, or too little difference to matter. 

173 NO_NEED = "no_need" 

174 #: For the house only: its windows look open, and the air coming in is getting damper. 

175 CLOSE = "close" 

176 #: For the house only: its windows look open, and its rooms already hold the outdoor air. 

177 AIRED = "aired"

180def airing(humidity_pct: float, after_pct: float | None) -> Airing: 

181 """Advice for one room, from its humidity and the humidity it would settle at once aired 

182 (None when that is unknown).""" 

183 if dries(humidity_pct, after_pct): 

184 return Airing.OPEN 

185 if _dampens(humidity_pct, after_pct): 

186 return Airing.KEEP_CLOSED 

187 return Airing.NO_NEED

190def dries(humidity_pct: float, after_pct: float | None) -> bool: 

191 """Whether airing would dry a room: one at least AIRING_HUMIDITY_PCT humid, brought at least 

192 DAMPER_BY_PCT points lower.""" 

193 return ( 

194 after_pct is not None 

195 and humidity_pct >= AIRING_HUMIDITY_PCT 

196 and after_pct <= humidity_pct - DAMPER_BY_PCT 

197 )

200def house_airing( 

201 rooms: list[tuple[float, float | None]], 

202 humidity_pct: float, 

203 after_pct: float | None, 

204 *, 

205 windows_open: bool = False, 

206 damper: bool = False, 

207) -> Airing: 

208 """Advice for the whole house, from each room's humidity now and once aired, the median 

209 room's humidity now and once aired, whether the windows look open (`windows_open`) and 

210 whether the air coming in is getting damper (`damper`). 

211 

212 With the windows open, close them once the air turns damper: the rooms follow it up, and 

213 by the time they read damp it is too late. Otherwise air when any room would be dried by it, 

214 and say the windows look open when the rooms already hold the outdoor air, where airing 

215 longer changes nothing. Keep the windows closed only when airing would leave the median room 

216 damp and damper than it is: when it would merely not help, or dampen dry rooms a little, 

217 nothing is lost by opening them. 

218 """ 

219 if windows_open and damper: 

220 return Airing.CLOSE 

221 if any(airing(*room) is Airing.OPEN for room in rooms): 

222 return Airing.OPEN 

223 if windows_open: 

224 return Airing.AIRED 

225 if _dampens(humidity_pct, after_pct): 

226 return Airing.KEEP_CLOSED 

227 return Airing.NO_NEED

230#: The windows look open when the house has read as aired -- its rooms sharing the outdoor air, 

231#: which the collector judges -- for at least this share of the last WINDOWS_OPEN_WITHIN. 

232WINDOWS_OPEN_SHARE = 0.75 

233WINDOWS_OPEN_WITHIN_MIN = 40 

234#: The air coming in is getting damper when the outdoor dew point has risen at least this much 

235#: in an hour... 

236DAMPER_DEWPOINT_RISE_C = 0.7 

237#: ...or the rooms are at least this many points above their lowest in DAMPER_LOW_WITHIN_H. The 

238#: Close the windows alert rule in grafana/alerts.json uses the same numbers. 

239DAMPER_HUMIDITY_RISE_PCT = 4.0 

240DAMPER_LOW_WITHIN_H = 2

243#: How far the inside of a window's glass sits from the outdoor temperature towards the room's:

249def glass_mists(room_temp_c: float, room_dewpoint_c: float, outdoor_c: float) -> bool: 

250 """Whether a room's window mists up with the air outside at `outdoor_c`: its glass falls to 

251 the room's dew point.""" 

252 glass = outdoor_c + GLASS_EDGE_FACTOR * (room_temp_c - outdoor_c) 

253 return room_dewpoint_c >= glass

256def windows_open(aired_share: float | None) -> bool: 

257 """Whether the windows look open, from the share of recent reports with the house aired.""" 

258 return aired_share is not None and aired_share >= WINDOWS_OPEN_SHARE

261def damper(dewpoint_rise_c: float | None, humidity_rise_pct: float | None) -> bool: 

262 """Whether the air coming in is getting damper: the outdoor dew point's rise over an hour, 

263 or the rooms' humidity above their recent lowest (None when unknown).""" 

264 return (dewpoint_rise_c is not None and dewpoint_rise_c >= DAMPER_DEWPOINT_RISE_C) or ( 

265 humidity_rise_pct is not None and humidity_rise_pct >= DAMPER_HUMIDITY_RISE_PCT 

266 )

269def _dampens(humidity_pct: float, after_pct: float | None) -> bool: 

270 """Whether airing would leave a room damp, and damper than it is.""" 

271 return ( 

272 after_pct is not None 

273 and after_pct >= AIRING_HUMIDITY_PCT 

274 and after_pct >= humidity_pct + DAMPER_BY_PCT 

275 )

278def room_flags( 

279 temp_c: float | None, humidity_pct: float | None, lowest_pct: float | None = None 

280) -> list[str]: 

281 """What is wrong with a room, if anything: cold, hot, damp or dry. 

282 

283 Damp is judged from `lowest_pct`, the room's lowest humidity over the last DAMP_FOR_H hours 

284 (None without readings through all of them): a shower or an evening's cooking passes, and 

285 only a room that stays damp is one mould grows in. 

286 """ 

287 flags = [] 

288 if temp_c is not None and temp_c < ROOM_COLD_C: 

289 flags.append("cold") 

290 if temp_c is not None and temp_c > ROOM_HOT_C: 

291 flags.append("hot") 

292 if lowest_pct is not None and lowest_pct >= ROOM_DAMP_PCT: 

293 flags.append("damp") 

294 if humidity_pct is not None and humidity_pct <= ROOM_DRY_PCT: 

295 flags.append("dry") 

296 return flags
