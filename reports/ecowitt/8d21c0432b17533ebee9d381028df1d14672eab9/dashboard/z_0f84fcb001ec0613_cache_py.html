1"""Answers kept for a short while, so a crowd of visitors costs the database one query each. 

2 

3Concurrent requests for the same key share one computation: the first starts it and the rest 

4wait on its result, so a burst of visitors arriving together does not become a burst of queries. 

5A failure is not kept; the next request tries again. 

6""" 

7 

8from __future__ import annotations 

9 

10import asyncio 

11import math 

12import time 

13from collections import OrderedDict 

14from collections.abc import Awaitable, Callable, Hashable 

15from dataclasses import dataclass, field 

16from typing import Any 

17 

18 

19@dataclass 

20class TtlCache: 

21 """At most `max_entries` results, each for as long as the caller asked.""" 

22 

23 max_entries: int = 1024 

24 clock: Callable[[], float] = time.monotonic 

25 _entries: OrderedDict[Hashable, tuple[float, asyncio.Future[Any]]] = field( 

26 default_factory=OrderedDict 

27 ) 

28 

29 async def get[T](self, key: Hashable, ttl: float, compute: Callable[[], Awaitable[T]]) -> T: 

30 """The cached result for `key`, or `compute()`'s, kept for `ttl` seconds. 

31 

32 The computation runs as its own task, so a visitor who disconnects while waiting does 

33 not cancel it for the others. 

34 """ 

35 entry = self._entries.get(key) 

36 if entry is not None and (not entry[1].done() or entry[0] > self.clock()): 

37 self._entries.move_to_end(key) 

38 return await asyncio.shield(entry[1]) 

39 task: asyncio.Future[T] = asyncio.ensure_future(compute()) 

40 self._entries[key] = (math.inf, task) 

41 while len(self._entries) > self.max_entries: 

42 self._entries.popitem(last=False) 

43 

44 def settle(done: asyncio.Future[T]) -> None: 

45 # Asking for the exception marks it retrieved even when no visitor is left waiting. 

46 failed = done.cancelled() or done.exception() is not None 

47 if self._entries.get(key, (0.0, None))[1] is not done: 

48 return 

49 if failed: 

50 del self._entries[key] 

51 else: 

52 self._entries[key] = (self.clock() + ttl, done) 

53 

54 task.add_done_callback(settle) 

55 return await asyncio.shield(task) 

56 

57 def clear(self) -> None: 

58 """Forget everything.""" 

59 self._entries.clear()
