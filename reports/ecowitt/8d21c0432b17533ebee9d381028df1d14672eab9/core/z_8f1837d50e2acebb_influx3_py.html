101 async def span( 

102 self, 

103 station: str, 

104 table: str, 

105 fields: Collection[str], 

106 *, 

107 start: datetime, 

108 end: datetime, 

109 sensors: Collection[str] | None = None, 

110 ) -> list[Span]: 

111 """Each group's first and last values in the window.""" 

112 info = await self._table(table) 

113 present = _present(fields, info.columns) 

114 if not present: 

115 return [] 

116 select = ["min(time) AS first_time", "max(time) AS last_time"] 

117 for i, name in enumerate(present): 

118 select.append(f'first_value("{name}" IGNORE NULLS ORDER BY time) AS first_{i}') 

119 select.append(f'last_value("{name}" IGNORE NULLS ORDER BY time) AS last_{i}') 

120 rows = await self._grouped(table, info, select, station, start, end, sensors) 

121 return [ 

122 Span( 

123 tags=_tags(row, info), 

124 name=row.get("name"), 

125 first_time=_time(row["first_time"]), 

126 last_time=_time(row["last_time"]), 

127 first=_values(row, "first", present), 

128 last=_values(row, "last", present), 

129 ) 

130 for row in rows 

131 if row.get("last_time") 

132 ]

134 async def extremes( 

135 self, 

136 station: str, 

137 table: str, 

138 fields: Collection[str], 

139 *, 

140 start: datetime, 

141 end: datetime, 

142 sensors: Collection[str] | None = None, 

143 ) -> list[Extremes]: 

144 """Each group's lowest and highest values in the window, with when each was first seen.""" 

145 info = await self._table(table) 

146 present = _present(fields, info.columns) 

147 if not present: 

148 return [] 

149 select = [] 

150 for i, name in enumerate(present): 

151 select += [ 

152 f'min("{name}") AS min_{i}', 

153 f'first_value(time ORDER BY "{name}" ASC NULLS LAST, time) AS min_{i}_at', 

154 f'max("{name}") AS max_{i}', 

155 f'first_value(time ORDER BY "{name}" DESC NULLS LAST, time) AS max_{i}_at', 

156 ] 

157 rows = await self._grouped(table, info, select, station, start, end, sensors) 

158 found = [] 

159 for row in rows: 

160 lows, highs = {}, {} 

161 for i, name in enumerate(present): 

162 if row.get(f"min_{i}") is not None: 

163 lows[name] = Extreme(float(row[f"min_{i}"]), _time(row[f"min_{i}_at"])) 

164 highs[name] = Extreme(float(row[f"max_{i}"]), _time(row[f"max_{i}_at"])) 

165 if lows: 

166 found.append(Extremes(_tags(row, info), row.get("name"), lows, highs)) 

167 return found

169 async def buckets( 

170 self, 

171 station: str, 

172 table: str, 

173 aggregates: Sequence[tuple[str, Aggregate]], 

174 *, 

175 start: datetime, 

176 end: datetime, 

177 step: timedelta, 

178 origin: datetime, 

179 sensors: Collection[str] | None = None, 

180 ) -> list[Buckets]: 

181 """Each group's values in buckets of `step` aligned to `origin`.""" 

182 info = await self._table(table) 

183 present = set(_present([name for name, _ in aggregates], info.columns)) 

184 wanted = list(dict.fromkeys(key for key in aggregates if key[0] in present)) 

185 if not wanted: 

186 return [] 

187 seconds = int(step.total_seconds()) 

188 if seconds <= 0: 

189 raise ValueError(f"a bucket must last at least a second, not {step}") 

190 bucket = f"date_bin(INTERVAL '{seconds} seconds', time, {_timestamp(origin)})" 

191 select = [f"{bucket} AS bucket"] 

192 for i, (name, aggregate) in enumerate(wanted): 

193 select.append(f"{_aggregate(name, aggregate)} AS value_{i}") 

194 rows = await self._grouped( 

195 table, info, select, station, start, end, sensors, extra_group="bucket" 

196 ) 

197 groups: dict[tuple[tuple[str, str], ...], Buckets] = {} 

198 names: dict[tuple[tuple[str, str], ...], str | None] = {} 

199 for row in rows: 

200 tags = _tags(row, info) 

201 key = tuple(sorted(tags.items())) 

202 if key not in groups: 

203 groups[key] = Buckets(tags, None, [], {pair: [] for pair in wanted}) 

204 # Rows arrive in time order, so the last bucket's name is the newest. 

205 names[key] = row.get("name") or names.get(key) 

206 groups[key].times.append(_time(row["bucket"])) 

207 for i, pair in enumerate(wanted): 

208 value = row.get(f"value_{i}") 

209 if value is not None and pair[1] is Aggregate.CIRCULAR: 

210 value = value % 360.0 

211 groups[key].values[pair].append(None if value is None else float(value)) 

212 return [ 

213 Buckets(group.tags, names[key], group.times, group.values) 

214 for key, group in groups.items() 

215 ]

217 async def _grouped( 

218 self, 

219 table: str, 

220 info: _Table, 

221 select: list[str], 

222 station: str, 

223 start: datetime, 

224 end: datetime, 

225 sensors: Collection[str] | None, 

226 *, 

227 extra_group: str | None = None, 

228 ) -> list[dict[str, Any]]: 

229 """One aggregate query over a station's rows in a window, grouped by the table's tags.""" 

230 params: dict[str, str] = {"station": station} 

231 where = [ 

232 "station = $station", 

233 f"time >= {_timestamp(start)}", 

234 f"time <= {_timestamp(end)}", 

235 ] 

236 group = info.groups 

237 if group: 

238 select = [*group, *select] 

239 if "name" in info.tags: 

240 select.append("last_value(name ORDER BY time) AS name") 

241 if sensors is not None and "sensor" in info.tags: 

242 if not sensors: 

243 return [] 

244 names = [] 

245 for i, sensor in enumerate(sorted(sensors)): 

246 params[f"s{i}"] = sensor 

247 names.append(f"$s{i}") 

248 where.append(f"sensor IN ({', '.join(names)})") 

249 if extra_group: 

250 group.append(extra_group) 

251 sql = f'SELECT {", ".join(select)} FROM "{table}" WHERE {" AND ".join(where)}' 

252 if group: 

253 sql += f" GROUP BY {', '.join(group)} ORDER BY {', '.join(group)}" 

254 return await self._rows(sql, params)

256 async def _table(self, table: str) -> _Table: 

257 """The columns of `table`, none if it does not exist; the catalogue is cached briefly. 

258 

259 InfluxDB 3 stores tags as dictionary-encoded strings, which is how they are told apart 

260 from fields. 

261 """ 

262 if not IDENTIFIER.fullmatch(table): 

263 raise ValueError(f"not a table name: {table!r}") 

264 now = time.monotonic() 

265 if now - self._columns_at > COLUMNS_TTL_SECONDS: 

266 rows = await self._rows( 

267 "SELECT table_name, column_name, data_type FROM information_schema.columns " 

268 "WHERE table_schema = 'iox'" 

269 ) 

270 columns: dict[str, set[str]] = {} 

271 tags: dict[str, list[str]] = {} 

272 for row in rows: 

273 columns.setdefault(row["table_name"], set()).add(row["column_name"]) 

274 if str(row.get("data_type", "")).startswith("Dictionary"): 

275 tags.setdefault(row["table_name"], []).append(row["column_name"]) 

276 self._tables = { 

277 name: _Table(frozenset(names), tuple(tags.get(name, ()))) 

278 for name, names in columns.items() 

279 } 

280 self._columns_at = now 

281 return self._tables.get(table, _Table())

290 async def _query( 

291 self, sql: str, params: dict[str, str] | None = None, *, timeout: float | None = None 

292 ) -> list[dict[str, Any]]: 

293 """POST one SQL query, returning its rows as dicts with null columns left out.""" 

294 if not self.configured: 

295 raise ReadError("no database is configured") 

296 body: dict[str, object] = {"db": self.database, "q": sql, "format": "json"} 

297 if params: 

298 body["params"] = params 

299 headers = {"Authorization": f"Bearer {self.token}"} if self.token else {} 

300 try: 

301 response = await self._client.post( 

302 self.url + "/api/v3/query_sql", 

303 json=body, 

304 headers=headers, 

305 timeout=timeout or TIMEOUT_SECONDS, 

306 ) 

307 except httpx2.RequestError as exc: 

308 raise ReadError(f"cannot connect ({type(exc).__name__})") from exc 

309 status = response.status_code 

310 if status == 200: 

311 rows = response.json() 

312 if not isinstance(rows, list): 

313 raise ReadError("InfluxDB answered with something other than rows") 

314 return rows 

315 text = response.text[:ERROR_EXCERPT] 

316 if status == 400 and re.search(r"table '[^']*' not found", text): 

317 raise _Missing(text) 

318 logger.error("query to InfluxDB failed: HTTP %s: %s", status, text) 

319 reasons = { 

320 401: "the token was not accepted", 

321 403: "the token may not read that database", 

322 404: "no such database", 

323 } 

324 raise ReadError(reasons.get(status, f"InfluxDB answered HTTP {status}"))

327def _present(fields: Collection[str], columns: frozenset[str]) -> list[str]: 

328 """The requested fields the table has, in order, refusing anything that is not a name.""" 

329 for name in fields: 

330 if not IDENTIFIER.fullmatch(name) or name in {"time", "station", "sensor", "name"}: 

331 raise ValueError(f"not a field name: {name!r}") 

332 return [name for name in dict.fromkeys(fields) if name in columns]

335def _tags(row: dict[str, Any], info: _Table) -> dict[str, str]: 

336 """A result row's group: its value of each tag the query grouped by.""" 

337 return {tag: row[tag] for tag in info.groups if row.get(tag) is not None}

340def _aggregate(name: str, aggregate: Aggregate) -> str: 

341 """The SQL that reduces one field's values in a bucket.""" 

342 match aggregate: 

343 case Aggregate.MEAN: 

344 return f'avg("{name}")' 

345 case Aggregate.MIN: 

346 return f'min("{name}")' 

347 case Aggregate.MAX: 

348 return f'max("{name}")' 

349 # Aggregate.CIRCULAR 

350 return f'degrees(atan2(avg(sin(radians("{name}"))), avg(cos(radians("{name}")))))'

353def _values(row: dict[str, Any], prefix: str, present: list[str]) -> dict[str, Value]: 

354 """The non-null `<prefix>_<i>` columns of a row, by field name.""" 

355 return { 

356 name: row[f"{prefix}_{i}"] 

357 for i, name in enumerate(present) 

358 if row.get(f"{prefix}_{i}") is not None 

359 }

362def _timestamp(moment: datetime) -> str: 

363 """A SQL timestamp literal for an aware datetime, in UTC to the second.""" 

364 if moment.tzinfo is None: 

365 raise ValueError("a query's times must carry a time zone") 

366 return f"TIMESTAMP '{moment.astimezone(UTC):%Y-%m-%dT%H:%M:%SZ}'"

369def _time(text: str) -> datetime: 

370 """A UTC datetime from InfluxDB's timestamps, which have no zone and up to nanoseconds.""" 

371 whole, _, fraction = text.partition(".") 

372 moment = datetime.fromisoformat(whole).replace(tzinfo=UTC) 

373 if fraction: 

374 moment += timedelta(microseconds=int(fraction[:6].ljust(6, "0"))) 

375 return moment
