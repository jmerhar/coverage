1"""Settings that describe the installation rather than its sensors. 

2 

3Infrastructure lives here, in the environment: where to listen, where InfluxDB is, how much 

4to log. Everything describing a particular station -- channel names, unit preferences, the 

5PASSKEY allowlist, the site location -- belongs to the station configuration file read by 

6`stationconfig`, because none of it is pleasant to express as environment variables. 

7 

8Every credential also accepts a `*_FILE` variant naming a file to read it from, which is what 

9makes a Docker secret usable without putting the value in a compose file. 

10""" 

11 

12from __future__ import annotations 

13 

14import functools 

15from pathlib import Path 

16from typing import Literal, Self 

17 

18from pydantic import field_validator, model_validator 

19from pydantic_settings import BaseSettings, SettingsConfigDict 

20 

21#: Maximum size of a report body. Real payloads are a few hundred bytes -- a full station 

22#: with every sensor type populated is still under 4 KB -- so this is generous by two orders 

23#: of magnitude while still bounding what an anonymous caller can make the process allocate. 

24MAX_BODY_BYTES = 64 * 1024 

25 

26#: Maximum number of form fields in a report, for the same reason. 

27MAX_BODY_FIELDS = 512 

28 

29 

30class Settings(BaseSettings): 

31 """Where this instance listens, and what it talks to.""" 

32 

33 model_config = SettingsConfigDict(env_file=".env", extra="ignore") 

34 

35 #: Holds the station configuration file and the write spool. 

36 data_dir: Path = Path("/data") 

37 

38 # Both listeners bind every interface because the shipped artefact is a container, where 

39 # a loopback bind is unreachable even through a published port. Which of them the 

40 # internet can reach is therefore decided by the host-side publish -- the ingest port on 

41 # a real address, the admin port on 127.0.0.1 -- not here. See the README. 

42 # 

43 # The defaults are the published ports, so host and container numbers match. Firewalls 

44 # that govern Docker traffic, ufw-docker among them, match the port after Docker's address 

45 # translation, which is the container's: a rule for 2551 does nothing for a container 

46 # listening on 8000 behind a 2551 publish. 

47 ingest_host: str = "0.0.0.0" # noqa: S104 

48 ingest_port: int = 2551 

49 admin_host: str = "0.0.0.0" # noqa: S104 

50 admin_port: int = 2552 

51 

52 #: The ingest listener's per-address budget: requests per second, and the burst allowed 

53 #: before that applies. A console reports at most once every 8 seconds. 

54 ingest_rate: float = 2.0 

55 ingest_burst: int = 20 

56 

57 #: The console's "Path" field. Ecowitt firmware sends it with a trailing slash; both 

58 #: spellings are served, so either value works here. 

59 ingest_path: str = "/data/report/" 

60 

61 influx_url: str = "" 

62 influx_database: str = "weather" 

63 influx_token: str = "" 

64 influx_token_file: Path | None = None 

65 #: InfluxDB 3 writes line protocol to /api/v3/write_lp; 2.x to /api/v2/write. 

66 influx_api: Literal["v3", "v2"] = "v3" 

67 #: Required by InfluxDB 2.x only, which scopes a bucket to an organisation. 

68 influx_org: str = "" 

69 

70 log_level: str = "INFO"

72 @field_validator("influx_token_file", mode="before") 

73 @classmethod 

74 def _blank_is_unset(cls, value: object) -> object: 

75 """Treat an empty value as absent rather than as a path. 

76 

77 Compose interpolates an unset `${VAR:-}` to an empty string and passes it to the 

78 container anyway, so an optional path arrives as "" rather than not arriving. Pydantic 

79 would resolve that to `Path(".")` -- the working directory -- and reading a credential 

80 from it fails with `IsADirectoryError` at startup. 

81 """ 

82 if isinstance(value, str) and not value.strip(): 

83 return None 

84 return value 

85 

86 @model_validator(mode="after") 

87 def _read_secret_files(self) -> Self: 

88 """Load any credential given as a file, and reject one that cannot be read. 

89 

90 Naming a file excludes the alternatives, so an unreadable one is an error rather than 

91 a fall back to the empty default -- which would otherwise surface much later as an 

92 authentication failure against InfluxDB. 

93 """ 

94 if self.influx_token_file is not None: 

95 self.influx_token = self.influx_token_file.read_text(encoding="utf-8").strip() 

96 return self 

97 

98 @model_validator(mode="after") 

99 def _normalise_ingest_path(self) -> Self: 

100 """Give the ingest path a leading slash, whatever was configured.""" 

101 if not self.ingest_path.startswith("/"): 

102 self.ingest_path = "/" + self.ingest_path 

103 return self 

104 

105 @property 

106 def ingest_paths(self) -> tuple[str, ...]: 

107 """Both slash spellings of the ingest path, in registration order. 

108 

109 Consoles differ over the trailing slash and Starlette's own redirect would answer the 

110 other spelling with a 307, which the Ecowitt uploader does not follow: it would retry 

111 the identical request for ever. Serving both outright is the only thing that works. 

112 """ 

113 bare = self.ingest_path.rstrip("/") 

114 if not bare: 

115 return ("/",) 

116 return (bare, bare + "/") 

117 

118 @property 

119 def config_file(self) -> Path: 

120 """The station configuration file.""" 

121 return self.data_dir / "config.yaml" 

122 

123 @property 

124 def spool_dir(self) -> Path: 

125 """Where writes wait when InfluxDB is unreachable.""" 

126 return self.data_dir / "spool"

129@functools.cache 

130def get_settings() -> Settings: 

131 """Return the process-wide settings, read from the environment once.""" 

132 return Settings()
