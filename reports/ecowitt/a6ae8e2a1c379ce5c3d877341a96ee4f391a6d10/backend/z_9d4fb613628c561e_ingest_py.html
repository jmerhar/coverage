1"""The public listener: one route, and nothing else. 

2 

3This is the only surface the open internet reaches, so it carries no status page, no API, no 

4setup wizard and no schema documentation -- not because those are guarded here, but because 

5they are not mounted here at all. Anything that is absent cannot be exposed by a mistake in a 

6guard. The admin listener in `admin` holds all of it. 

7"""

27#: What the Ecowitt cloud answers an upload with. The firmware does not inspect it closely, 

28#: but it is the reply a station is built to expect, and it reveals nothing about this server. 

29OK_BODY = {"errcode": "0", "errmsg": "ok"} 

30 

31#: Never logged. PASSKEY identifies an Ecowitt-protocol station and authenticates its reports; 

32#: PASSWORD is the Wunderground protocol's equivalent. A log file is the easiest place for a 

33#: credential to be read from. 

34REDACTED_FIELDS = frozenset({"PASSKEY", "PASSWORD"}) 

35

37class ReportHandler(Protocol): 

38 """What the listener does with a report once it has been read and split into fields.""" 

39 

40 async def handle(self, fields: Mapping[str, str], source: str) -> bool: 

41 """Process one report, returning whether it was accepted.""" 

42

44def redact(fields: Mapping[str, str]) -> dict[str, str]: 

45 """Copy the fields with every credential replaced, safe to log.""" 

46 return {k: ("<redacted>" if k in REDACTED_FIELDS else v) for k, v in fields.items()}

49def build_app( 

50 settings: Settings, 

51 state: State, 

52 handler: ReportHandler, 

53 limiter: RateLimiter | None = None, 

54) -> FastAPI: 

55 """Build the public application, serving only the configured ingest path.""" 

56 limiter = limiter or RateLimiter(rate=settings.ingest_rate, burst=settings.ingest_burst) 

57 app = FastAPI( 

58 title="Ecowitt Server ingest", 

59 # No interactive docs and no schema: they would describe this server to anyone who 

60 # found the port. 

61 docs_url=None, 

62 redoc_url=None, 

63 openapi_url=None, 

64 # Starlette's slash redirect answers with a 307, which the Ecowitt uploader does not 

65 # follow -- it would retry the same request for ever. Both spellings are registered 

66 # explicitly instead. 

67 redirect_slashes=False, 

68 )

70 async def report(request: Request) -> Response: 

71 """Receive one station report.""" 

72 source = request.client.host if request.client else "unknown" 

73 

74 # Before the body is read, so an address over its budget costs almost nothing. 

75 if not limiter.allow(source): 

76 state.record_rate_limited() 

77 return PlainTextResponse("", status_code=429) 

78 

79 try: 

80 body = await read_capped_body(request, MAX_BODY_BYTES) 

81 except BodyTooLarge: 

82 logger.warning("report from %s rejected: body over %d bytes", source, MAX_BODY_BYTES) 

83 state.record_rejected() 

84 return PlainTextResponse("", status_code=413) 

85 

86 # The Ecowitt protocol posts form-encoded fields; the Wunderground variant puts the 

87 # same names in the query string, and both are accepted. parse_qsl is total -- it 

88 # yields what it can and never raises -- so a malformed body degrades to fewer fields 

89 # rather than to a 500. 

90 pairs = parse_qsl(body.decode("utf-8", errors="replace"), keep_blank_values=True) 

91 pairs += parse_qsl(request.url.query, keep_blank_values=True) 

92 if len(pairs) > MAX_BODY_FIELDS: 

93 logger.warning("report from %s rejected: over %d fields", source, MAX_BODY_FIELDS) 

94 state.record_rejected() 

95 return PlainTextResponse("", status_code=413) 

96 

97 if await handler.handle(dict(pairs), source): 

98 state.record_accepted() 

99 else: 

100 state.record_rejected() 

101 

102 # Always 200, whether or not the report was kept. A station cannot act on a refusal -- 

103 # it has no backlog to retry from -- and a distinguishable error would tell an 

104 # anonymous caller which PASSKEY guesses were wrong. 

105 return JSONResponse(OK_BODY) 

106 

107 for path in settings.ingest_paths: 

108 app.add_api_route(path, report, methods=["POST", "GET"], include_in_schema=False) 

109 

110 @app.exception_handler(404) 

111 async def not_found(_request: Request, _exc: Exception) -> Response: 

112 """Answer anything that is not the ingest path with an empty 404.""" 

113 return PlainTextResponse("", status_code=404) 

114 

115 return app

118class LoggingHandler: 

119 """A report handler that only records what it saw. 

120 

121 Useful on its own for confirming a console can reach this server, and as the stand-in in 

122 tests that exercise the listener rather than the storage path. 

123 """ 

124 

125 def __init__(self) -> None: 

126 self.reports: list[dict[str, str]] = [] 

127 

128 async def handle(self, fields: Mapping[str, str], source: str) -> bool: 

129 """Record the report and accept it.""" 

130 self.reports.append(dict(fields)) 

131 logger.info("report from %s: %d fields", source, len(fields)) 

132 logger.debug("report from %s: %s", source, redact(fields)) 

133 return True
