1"""Send line protocol to InfluxDB. 

2 

3InfluxDB 3 takes it at `/api/v3/write_lp` with a bearer token; 2.x at `/api/v2/write`, where a 

4database is a bucket inside an organisation and the token scheme is `Token`. Both are asked for 

5second precision, which is what the encoder writes. 

6""" 

7 

8from __future__ import annotations 

9 

10import logging 

11from enum import Enum 

12 

13import httpx2 

14 

15from .config import Settings 

16 

17logger = logging.getLogger(__name__) 

18 

19TIMEOUT_SECONDS = 10.0 

20#: How much of an error response to log. InfluxDB explains a rejected write in the body, which 

21#: is the useful part, but a proxy's HTML error page is not worth a screenful. 

22ERROR_EXCERPT = 300 

23 

24 

25class Outcome(Enum): 

26 """What became of one attempt to write a report.""" 

27 

28 #: Stored. 

29 OK = "ok" 

30 #: Not stored, and worth trying again: the database is unreachable, overloaded, or 

31 #: refusing this client for a reason someone can fix. 

32 RETRY = "retry" 

33 #: Not stored, and never will be: InfluxDB refused the data itself. Retrying would block 

34 #: every report queued behind this one. 

35 REJECT = "reject" 

36 

37 

38#: Client errors that describe the server or the credentials rather than the data. A revoked 

39#: token or a dropped database is a fault to fix, and the readings should still be there once it 

40#: is: 401 and 403 are authentication, 404 a missing database, 408 and 429 load. 

41RETRYABLE_CLIENT_ERRORS = frozenset({401, 403, 404, 408, 429}) 

42 

43 

44def classify(status: int) -> Outcome: 

45 """The outcome an HTTP status means for a write.""" 

46 if 200 <= status < 300: 

47 return Outcome.OK 

48 if status in RETRYABLE_CLIENT_ERRORS or status >= 500: 

49 return Outcome.RETRY 

50 return Outcome.REJECT 

51 

52 

53class InfluxWriter: 

54 """Writes encoded rows to one InfluxDB database.""" 

55 

56 def __init__(self, settings: Settings, client: httpx2.AsyncClient | None = None) -> None: 

57 self._settings = settings 

58 self._client = client or httpx2.AsyncClient(timeout=TIMEOUT_SECONDS) 

59 self.last_error: str | None = None 

60 

61 @property 

62 def configured(self) -> bool: 

63 """Whether there is anywhere to write to.""" 

64 return bool(self._settings.influx_url) 

65 

66 async def send(self, body: str) -> Outcome: 

67 """Write rows once, returning what became of them. 

68 

69 Failures are logged and reported rather than raised: the station's reply never waits 

70 on, or depends on, the database. 

71 """ 

72 if not body: 

73 return Outcome.OK 

74 if not self.configured: 

75 return self._fail(Outcome.RETRY, "INFLUX_URL is not set") 

76 url, params, headers = self._request() 

77 try: 

78 response = await self._client.post( 

79 url, params=params, headers=headers, content=body.encode("utf-8") 

80 ) 

81 except httpx2.RequestError as exc: 

82 return self._fail(Outcome.RETRY, f"{type(exc).__name__}: {exc}") 

83 outcome = classify(response.status_code) 

84 if outcome is not Outcome.OK: 

85 detail = f"HTTP {response.status_code}: {response.text[:ERROR_EXCERPT]}" 

86 return self._fail(outcome, detail) 

87 self.last_error = None 

88 return Outcome.OK 

89 

90 async def aclose(self) -> None: 

91 """Release the HTTP client's connections.""" 

92 await self._client.aclose() 

93 

94 def _request(self) -> tuple[str, dict[str, str], dict[str, str]]: 

95 """The URL, query and headers for this server's API version.""" 

96 s = self._settings 

97 base = s.influx_url.rstrip("/") 

98 headers = {"Content-Type": "text/plain; charset=utf-8"} 

99 if s.influx_api == "v2": 

100 if s.influx_token: 

101 headers["Authorization"] = f"Token {s.influx_token}" 

102 params = {"bucket": s.influx_database, "org": s.influx_org, "precision": "s"} 

103 return f"{base}/api/v2/write", params, headers 

104 if s.influx_token: 

105 headers["Authorization"] = f"Bearer {s.influx_token}" 

106 params = {"db": s.influx_database, "precision": "second"} 

107 return f"{base}/api/v3/write_lp", params, headers 

108 

109 def _fail(self, outcome: Outcome, reason: str) -> Outcome: 

110 """Record and log a failed write.""" 

111 self.last_error = reason 

112 verdict = "will retry" if outcome is Outcome.RETRY else "refused, not retrying" 

113 logger.error("write to InfluxDB failed (%s): %s", verdict, reason) 

114 return outcome
