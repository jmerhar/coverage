1"""Psychrometric and barometric formulas, in canonical units (°C, %, hPa, m). 

2 

3Pure functions with no knowledge of reports, so each can be checked against published 

4reference values on its own. 

5""" 

6 

7from __future__ import annotations 

8 

9import math 

10 

11# Magnus coefficients over water from Alduchov & Eskridge (1996), accurate to within 0.4% of 

12# the Goff-Gratch reference between -40 °C and 50 °C -- the range a weather station lives in. 

13MAGNUS_A = 6.1094 

14MAGNUS_B = 17.625 

15MAGNUS_C = 243.04 

16 

17#: Specific gas constant of water vapour, J/(kg·K). 

18R_VAPOUR = 461.5 

19#: Ratio of the molar masses of water and dry air, times 1000 for grams per kilogram. 

20EPSILON_G_PER_KG = 621.97 

21ZERO_CELSIUS_K = 273.15 

22 

23#: Temperature lapse rate of the international standard atmosphere, K/m. 

24LAPSE_RATE = 0.0065 

25#: Exponent g·M/(R·L) of the barometric formula for that lapse rate. 

26BAROMETRIC_EXPONENT = 5.257 

27 

28 

29def saturation_vapour_pressure(temp_c: float) -> float: 

30 """Vapour pressure of saturated air at `temp_c`, in hPa.""" 

31 return MAGNUS_A * math.exp(MAGNUS_B * temp_c / (temp_c + MAGNUS_C)) 

32 

33 

34def vapour_pressure(temp_c: float, humidity_pct: float) -> float: 

35 """Partial pressure of the water vapour actually present, in hPa.""" 

36 return humidity_pct / 100 * saturation_vapour_pressure(temp_c) 

37 

38 

39def dew_point(temp_c: float, humidity_pct: float) -> float: 

40 """Temperature to which air must cool for its vapour to condense, in °C. 

41 

42 The quantity that decides whether ventilating dries a room: outdoor air only takes moisture 

43 out if its dew point is below the indoor one. Relative humidity cannot answer that, because 

44 it depends on temperature as much as on how much water the air holds. 

45 """ 

46 gamma = math.log(humidity_pct / 100) + MAGNUS_B * temp_c / (temp_c + MAGNUS_C) 

47 return MAGNUS_C * gamma / (MAGNUS_B - gamma) 

48 

49 

50def absolute_humidity(temp_c: float, humidity_pct: float) -> float: 

51 """Mass of water vapour per volume of air, in g/m³.""" 

52 pascals = vapour_pressure(temp_c, humidity_pct) * 100 

53 return pascals / (R_VAPOUR * (temp_c + ZERO_CELSIUS_K)) * 1000 

54 

55 

56def mixing_ratio(temp_c: float, humidity_pct: float, pressure_hpa: float) -> float: 

57 """Mass of water vapour per mass of dry air, in g/kg. 

58 

59 Unlike absolute humidity this does not change as air warms or expands, which makes it the 

60 measure to compare between a room and the outside. 

61 """ 

62 e = vapour_pressure(temp_c, humidity_pct) 

63 return EPSILON_G_PER_KG * e / (pressure_hpa - e) 

64 

65 

66def vapour_pressure_deficit(temp_c: float, humidity_pct: float) -> float: 

67 """How far the air is from saturation, in hPa: the drying power of the air.""" 

68 return saturation_vapour_pressure(temp_c) - vapour_pressure(temp_c, humidity_pct) 

69 

70 

71def humidity_at(vapour_pressure_hpa: float, temp_c: float) -> float: 

72 """Relative humidity that air holding this much vapour would have at `temp_c`, in %. 

73 

74 Capped at 100: beyond it the excess condenses rather than raising the humidity. 

75 """ 

76 return min(100.0, vapour_pressure_hpa / saturation_vapour_pressure(temp_c) * 100)

79def sea_level_pressure(station_hpa: float, altitude_m: float, temp_c: float) -> float: 

80 """Reduce a station pressure to sea level, in hPa. 

81 

82 The barometric formula with the standard lapse rate, taking the column of air below the 

83 station to be as warm at its base as the standard atmosphere would make it given `temp_c` 

84 at the top. A console's relative-pressure offset approximates this with a constant, which 

85 is exact only at one temperature: the result moves by about 0.1 hPa per degree at 250 m. 

86 """ 

87 lapse = LAPSE_RATE * altitude_m 

88 return station_hpa * (1 - lapse / (temp_c + lapse + ZERO_CELSIUS_K)) ** -BAROMETRIC_EXPONENT 

89 

90 

91def standard_temperature(altitude_m: float) -> float: 

92 """Temperature of the international standard atmosphere at `altitude_m`, in °C.""" 

93 return 15.0 - LAPSE_RATE * altitude_m
