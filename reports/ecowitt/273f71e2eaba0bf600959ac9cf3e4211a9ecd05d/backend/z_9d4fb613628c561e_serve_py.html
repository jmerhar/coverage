37class _Listener(uvicorn.Server): 

38 """A uvicorn server that leaves the process's signal handlers alone.""" 

39 

40 @contextlib.contextmanager 

41 def capture_signals(self) -> Iterator[None]: 

42 """Install nothing, so one handler can serve both listeners. 

43 

44 `uvicorn.Server.serve` wraps itself in this, and the base implementation points 

45 SIGINT and SIGTERM at its own `handle_exit` with `signal.signal` -- which keeps a 

46 single handler per signal. With two servers in one process the second would displace 

47 the first, so a SIGTERM would stop only one of them and the process would hang in 

48 `asyncio.gather` until something sent SIGKILL. `run` installs one handler that stops 

49 every listener instead. 

50 """ 

51 yield

53 async def notify_serving(self, state: State, attribute: str) -> None: 

54 """Set `attribute` on `state` once this listener is accepting connections.""" 

55 while not self.started: 

56 await asyncio.sleep(0.05) 

57 setattr(state, attribute, True)

60#: Libraries that log every request at INFO. A station reports every few seconds, so at the 

61#: default level each would add a line per report, for ever; their warnings still get through. 

62CHATTY_LOGGERS = ("httpx2", "httpcore2")

65def configure_logging(level: str) -> None: 

66 """Send the application's logs to stderr at the configured level.""" 

67 logging.basicConfig( 

68 level=level.upper(), 

69 format="%(asctime)s %(levelname)-8s %(name)s %(message)s", 

70 ) 

71 # From the argument rather than read back from the root logger: basicConfig does nothing 

72 # when the root logger already has handlers, so the root level is not a reliable echo. 

73 configured = logging.getLevelNamesMapping()[level.upper()] 

74 for name in CHATTY_LOGGERS: 

75 logging.getLogger(name).setLevel(max(logging.WARNING, configured))

78def stop_all(listeners: Iterable[_Listener]) -> None: 

79 """Ask every listener to finish serving.""" 

80 logger.info("shutting down") 

81 for listener in listeners: 

82 listener.should_exit = True

85#: Most a request's line and headers may take together. A console's request is a few hundred 

86#: bytes; the cap stops an anonymous caller making the process buffer megabytes of headers, 

87#: which happens before the application, its body cap or its rate limit sees the request. 

88MAX_HEADER_BYTES = 16 * 1024 

89 

90#: Most connections each listener holds at once; beyond it, new requests get a 503. A station 

91#: holds one for a moment per report, so this bounds what slow or idle callers can pin open. 

92MAX_CONNECTIONS = 64

95def _limits() -> dict[str, object]: 

96 """Connection and header bounds shared by both listeners. 

97 

98 h11 rather than httptools, because only h11 has a limit on incomplete headers. 

99 """ 

100 return { 

101 "http": "h11", 

102 "h11_max_incomplete_event_size": MAX_HEADER_BYTES, 

103 "limit_concurrency": MAX_CONNECTIONS, 

104 }

107def build( 

108 settings: Settings, 

109 handler: ReportHandler, 

110 state: State, 

111 context: AdminContext | None = None, 

112) -> tuple[_Listener, _Listener]: 

113 """Build the two listeners. Without an admin context, the admin pages edit nothing.""" 

114 ingest_listener = _Listener( 

115 uvicorn.Config( 

116 app=ingest.build_app(settings, state, handler), 

117 host=settings.ingest_host, 

118 port=settings.ingest_port, 

119 # No `server: uvicorn` header, so the port does not name its software. 

120 server_header=False, 

121 # No access log: this listener answers the open internet, and a request line 

122 # would copy the ingest path -- which may carry a secret segment -- into the log 

123 # on every report. The handler logs an accepted report itself. 

124 access_log=False, 

125 log_config=None, 

126 **_limits(), # type: ignore[arg-type] 

127 ) 

128 ) 

129 admin_listener = _Listener( 

130 uvicorn.Config( 

131 app=admin.build_app(context or AdminContext(settings, state)), 

132 host=settings.admin_host, 

133 port=settings.admin_port, 

134 log_config=None, 

135 **_limits(), # type: ignore[arg-type] 

136 ) 

137 ) 

138 return ingest_listener, admin_listener

141def warn_if_admin_unauthenticated(settings: Settings, login_set: bool = False) -> bool: 

142 """Say so when the admin listener has no login, returning whether it warned. 

143 

144 The bind address cannot tell whether that matters. The shipped artefact is a container, 

145 where a loopback bind is unreachable even through a published port, so the admin listener 

146 always binds every interface; what keeps it private is where the host publishes it, which 

147 the process cannot see. Whether a login is set is the one thing it can see, so that is what 

148 the warning keys on. 

149 """ 

150 if login_set: 

151 return False 

152 logger.warning( 

153 "the admin interface has no login: publish port %d on loopback only, behind a reverse " 

154 "proxy, or set a login on the setup page", 

155 settings.admin_port, 

156 ) 

157 return True

160async def run(settings: Settings | None = None, handler: ReportHandler | None = None) -> None: 

161 """Serve until a signal arrives. 

162 

163 Without a handler given, reports are authenticated against the configuration file and 

164 delivered to InfluxDB, through the spool when it cannot take them. A configuration file 

165 that exists but cannot be read stops startup with the reason, rather than serving with no 

166 stations and discarding every report. 

167 """ 

168 settings = settings or get_settings() 

169 state = State() 

170 background: list[asyncio.Task[None]] = [] 

171 closers: list[Callable[[], Awaitable[None]]] = [] 

172 context: AdminContext | None = None 

173 if handler is None: 

174 store = ConfigStore(settings.config_file) 

175 writer = InfluxWriter(settings) 

176 lookup_client = httpx2.AsyncClient() 

177 closers += [writer.aclose, lookup_client.aclose] 

178 spool = Spool(settings.spool_dir, settings.spool_max_bytes) 

179 heartbeat = None 

180 if settings.heartbeat_url: 

181 heartbeat = Heartbeat( 

182 lookup_client, 

183 settings.heartbeat_url, 

184 interval_seconds=settings.heartbeat_interval_seconds, 

185 ) 

186 logger.info("heartbeat to %s after each write", heartbeat.host) 

187 delivery = Delivery( 

188 writer, spool, state, on_written=heartbeat.beat if heartbeat else lambda: None 

189 ) 

190 secret = auth.load_secret(settings.secret_file) 

191 pending, calibration = PendingStations(key=secret), CalibrationMonitor() 

192 station_handler = StationHandler( 

193 store.stations, delivery, pending=pending, calibration=calibration 

194 ) 

195 store.subscribe(lambda stations: setattr(station_handler, "config", stations)) 

196 handler = station_handler 

197 context = AdminContext( 

198 settings, 

199 state, 

200 secret=secret, 

201 store=store, 

202 handler=station_handler, 

203 pending=pending, 

204 calibration=calibration, 

205 spool=spool, 

206 http=lookup_client, 

207 ) 

208 reference = ReferenceUpdater( 

209 lambda: store.stations, 

210 calibration, 

211 lookup_client, 

212 url=settings.open_meteo_url, 

213 interval_seconds=settings.reference_interval_seconds, 

214 ) 

215 store.subscribe(lambda _: reference.wake()) 

216 calibration.on_step = reference.wake 

217 logger.info("stations: %s", ", ".join(store.stations.names) or "none configured") 

218 if not writer.configured: 

219 logger.warning("INFLUX_URL is not set: reports will be spooled until it is") 

220 background += [ 

221 asyncio.create_task(delivery.run(), name="spool-replay"), 

222 asyncio.create_task(reference.run(), name="reference-pressure"), 

223 ] 

224 if heartbeat is not None: 

225 background.append(asyncio.create_task(heartbeat.run(), name="heartbeat")) 

226 ingest_listener, admin_listener = build(settings, handler, state, context) 

227 

228 warn_if_admin_unauthenticated( 

229 settings, context is not None and context.document.admin is not None 

230 ) 

231 

232 listeners = (ingest_listener, admin_listener) 

233 loop = asyncio.get_running_loop() 

234 for sig in (signal.SIGINT, signal.SIGTERM): 

235 loop.add_signal_handler(sig, stop_all, listeners) 

236 

237 logger.info( 

238 "ingest on %s:%d%s, admin on %s:%d", 

239 settings.ingest_host, 

240 settings.ingest_port, 

241 settings.ingest_path, 

242 settings.admin_host, 

243 settings.admin_port, 

244 ) 

245 try: 

246 await asyncio.gather( 

247 ingest_listener.serve(), 

248 admin_listener.serve(), 

249 ingest_listener.notify_serving(state, "ingest_serving"), 

250 admin_listener.notify_serving(state, "admin_serving"), 

251 ) 

252 finally: 

253 for task in background: 

254 task.cancel() 

255 with contextlib.suppress(asyncio.CancelledError): 

256 await task 

257 for close in closers: 

258 await close() 

259

261def main() -> None: 

262 """Configure logging and serve.""" 

263 settings = get_settings() 

264 configure_logging(settings.log_level) 

265 asyncio.run(run(settings))
