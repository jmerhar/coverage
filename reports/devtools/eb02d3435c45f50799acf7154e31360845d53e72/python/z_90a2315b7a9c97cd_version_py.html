1#!/usr/bin/env python3 

2"""The release version, which every manifest, CHANGELOG.md and the compose file carry. 

3 

4 version.py check [TAG] every manifest names the same version, and the compose file pins each 

5 image to that version's line (see `line`); with a tag (v1.2.3), that 

6 the manifests name that version and CHANGELOG.md has its section 

7 version.py set X.Y.Z write X.Y.Z, which must follow the current version, into every 

8 manifest and its line into the compose file, and date CHANGELOG.md's 

9 Unreleased section as X.Y.Z's; a release that cannot be cut changes 

10 nothing 

11 version.py notes X.Y.Z print X.Y.Z's section of CHANGELOG.md, without its heading 

12 

13What carries the version is listed in devtools.toml: 

14 

15 [version] 

16 pyprojects = ["backend/pyproject.toml"] # each one's `version = "..."` line 

17 npm = ["frontend"] # package.json, and package-lock.json when present 

18 images = ["ghcr.io/jmerhar/myapp"] # optional: images the compose file pins 

19 compose = "docker-compose.yml" # optional, the default 

20 

21Between releases a manifest may carry a development version (0.2.0.dev0), which counts as coming 

22before the release it names. npm only accepts semver, so package.json and its lockfile spell the 

23same version 0.2.0-dev.0. `--root DIR` works on a checkout other than the current directory. 

24"""

26from __future__ import annotations

28import argparse 

29import importlib.util 

30import re 

31import sys 

32from dataclasses import dataclass, field 

33from datetime import date

36CHANGELOG = "CHANGELOG.md" 

37RELEASE = re.compile(r"\d+\.\d+\.\d+") 

38DEVELOPMENT = re.compile(r"(\d+\.\d+\.\d+)\.dev\d+") 

39NPM_DEVELOPMENT = re.compile(r"(\d+\.\d+\.\d+)-dev\.(\d+)") 

40UNRELEASED = "## Unreleased" 

41UNRELEASED_HEADING = re.compile(r"(?m)^## Unreleased$") 

42PYPROJECT_VERSION = re.compile(r'(?m)^(version = ")([^"]*)(")$') 

43#: package.json's version, and in package-lock.json both the top-level one and its root 

44#: package's; the dependencies' versions are left alone. 

45PACKAGE_VERSION = re.compile(r'(?m)^( "version": ")([^"]*)(")') 

46LOCK_ROOT_VERSION = re.compile( 

47 r'(?m)^( "": \{\n(?: "name": "[^"]*",\n)? "version": ")([^"]*)(")'

51class VersionError(Exception): 

52 """What stops a check or a release, said to the person running it."""

55@dataclass 

56class Layout: 

57 """Where a repository carries its version, as devtools.toml declares it.""" 

58 

59 #: Each manifest and the patterns of its own version lines. 

60 manifests: dict[str, list[re.Pattern[str]]] = field(default_factory=dict) 

61 images: list[str] = field(default_factory=list) 

62 compose: str = "docker-compose.yml"

65def _config_reader(): 

66 """Import devtools-config.py, which sits beside this script.""" 

67 path = Path(__file__).with_name("devtools-config.py") 

68 spec = importlib.util.spec_from_file_location("devtools_config", path) 

69 assert spec and spec.loader 

70 module = importlib.util.module_from_spec(spec) 

71 spec.loader.exec_module(module) 

72 return module

75def layout(root: Path) -> Layout: 

76 """Read the [version] table of the repository's devtools.toml. 

77 

78 Args: 

79 root: The repository root. 

80 

81 Returns: 

82 The manifests, images and compose file to keep in step. 

83 

84 Raises: 

85 VersionError: If devtools.toml is missing, invalid, or lists no manifest. 

86 """ 

87 reader = _config_reader() 

88 path = str(root / "devtools.toml") 

89 try: 

90 pyprojects = reader.lookup(path, "version.pyprojects", [], "list") 

91 npm = reader.lookup(path, "version.npm", [], "list") 

92 images = reader.lookup(path, "version.images", [], "list") 

93 compose = reader.lookup(path, "version.compose", ["docker-compose.yml"]) 

94 except reader.ConfigError as e: 

95 raise VersionError(str(e)) from None 

96 if len(compose) != 1: 

97 raise VersionError('version.compose must be one file, such as "docker-compose.yml"') 

98 [compose] = compose 

99 found = Layout(images=images, compose=compose) 

100 for pyproject in pyprojects: 

101 found.manifests[pyproject] = [PYPROJECT_VERSION] 

102 for directory in npm: 

103 found.manifests[f"{directory}/package.json"] = [PACKAGE_VERSION] 

104 lock = f"{directory}/package-lock.json" 

105 if (root / lock).is_file(): 

106 found.manifests[lock] = [PACKAGE_VERSION, LOCK_ROOT_VERSION] 

107 if not found.manifests: 

108 raise VersionError("devtools.toml lists no manifests; set [version] pyprojects or npm") 

109 return found

112def _read(root: Path, path: str) -> str: 

113 try: 

114 return (root / path).read_text(encoding="utf-8") 

115 except FileNotFoundError: 

116 raise VersionError(f"{path} is missing") from None

119def normalise(version: str) -> str: 

120 """The Python spelling of a version: npm's 0.2.0-dev.0 is 0.2.0.dev0.""" 

121 npm = NPM_DEVELOPMENT.fullmatch(version) 

122 return f"{npm.group(1)}.dev{npm.group(2)}" if npm else version

125def versions(root: Path, where: Layout) -> dict[str, str]: 

126 """Every manifest's version, by where it was found, in its Python spelling.""" 

127 found = {} 

128 for path, patterns in where.manifests.items(): 

129 text = _read(root, path) 

130 for n, pattern in enumerate(patterns): 

131 match = pattern.search(text) 

132 if match is None: 

133 raise VersionError(f"{path} names no version where one is expected") 

134 found[f"{path}#{n}" if n else path] = normalise(match.group(2)) 

135 return found

138def _order(version: str) -> tuple[int, ...]: 

139 """A key that sorts versions, a development version just before the release it names.""" 

140 if RELEASE.fullmatch(version): 

141 return (*map(int, version.split(".")), 1) 

142 development = DEVELOPMENT.fullmatch(version) 

143 if development: 

144 return (*map(int, development.group(1).split(".")), 0) 

145 raise VersionError(f"{version} is neither a release like 1.2.3 nor a development version")

148def line(version: str) -> str: 

149 """The image tag the compose file follows for `version`: its major, or before 1.0 its minor. 

150 

151 A 0.x release may change anything, and an image builder publishes no bare major tag for one, 

152 so until 1.0 the compose file follows major.minor instead. A development version follows the 

153 release it leads up to. 

154 """ 

155 development = DEVELOPMENT.fullmatch(version) 

156 major, minor, _ = (development.group(1) if development else version).split(".") 

157 return major if major != "0" else f"{major}.{minor}"

160def _pins(text: str, where: Layout) -> list[tuple[int, re.Match[str]]]: 

161 """Each line of the compose file naming one of the images, read for its tag.""" 

162 names = "|".join(re.escape(image) for image in where.images) 

163 ours = re.compile(rf"""^\s*image:\s*["']?(?:{names})(?:[:@"'\s]|$)""") 

164 # A tag followed by a digest is not a plain tag: a new version would keep the old digest. 

165 pin = re.compile(rf"""^(\s*image:\s*["']?(?:{names}):)([0-9A-Za-z._-]+)([^@]*)$""") 

166 pins = [] 

167 # Split on newlines only, as set_version does when it rewrites the lines by number. 

168 for n, text_line in enumerate(text.split("\n")): 

169 if ours.match(text_line): 

170 match = pin.match(text_line) 

171 if match is None: 

172 raise VersionError( 

173 f"{where.compose} line {n + 1} names an image without a plain tag" 

174 ) 

175 pins.append((n, match)) 

176 missing = [ 

177 image for image in where.images if not any(image + ":" in m.group(1) for _, m in pins) 

178 ] 

179 if missing: 

180 raise VersionError(f"{where.compose} does not name {', '.join(missing)}") 

181 return pins

184def check(root: Path, tag: str | None, where: Layout | None = None) -> str: 

185 """The version every manifest agrees on; with `tag`, also that it is the tag's, released.""" 

186 where = where or layout(root) 

187 found = versions(root, where) 

188 if len(set(found.values())) != 1: 

189 listed = "\n".join(f" {path}: {v}" for path, v in found.items()) 

190 raise VersionError(f"the manifests name different versions:\n{listed}") 

191 version = next(iter(found.values())) 

192 _order(version) 

193 if where.images: 

194 expected = line(version) 

195 for _, pin in _pins(_read(root, where.compose), where): 

196 if pin.group(2) != expected: 

197 image = pin.group(1).split("image:")[1].strip().strip("\"'") 

198 raise VersionError( 

199 f"{where.compose} pins {image}{pin.group(2)}, not {expected} as {version} is" 

200 ) 

201 if tag is not None: 

202 if tag != f"v{version}": 

203 raise VersionError(f"the tag is {tag}, but the manifests name {version}") 

204 notes(root, version) 

205 return version

208def set_version(root: Path, version: str, today: date) -> None: 

209 """Write `version` into every manifest and its line into the compose file, and date the 

210 Unreleased changelog section as it. 

211 

212 Every file is worked out before any is written, so a release that cannot be cut changes 

213 nothing. 

214 """ 

215 if not RELEASE.fullmatch(version): 

216 raise VersionError(f"{version} is not a version like 1.2.3") 

217 where = layout(root) 

218 current = check(root, None, where) 

219 if _order(version) <= _order(current): 

220 raise VersionError(f"{version} is not after {current}, the current version") 

221 text = _read(root, CHANGELOG) 

222 headings = UNRELEASED_HEADING.findall(text) 

223 if len(headings) != 1: 

224 found = "no" if not headings else "more than one" 

225 raise VersionError(f"{CHANGELOG} has {found} '{UNRELEASED}' section to release") 

226 if not any( 

227 line.strip() and not line.startswith("#") 

228 for line in _section(text, UNRELEASED).splitlines() 

229 ): 

230 raise VersionError(f"{CHANGELOG}'s '{UNRELEASED}' section is empty") 

231 if re.search(rf"(?m)^## {re.escape(version)}(\s|$)", text): 

232 raise VersionError(f"{CHANGELOG} already has a section for {version}") 

233 planned = {CHANGELOG: UNRELEASED_HEADING.sub(f"## {version} — {today.isoformat()}", text, 1)} 

234 # check() above found every pattern in its file, so each substitution has its line. 

235 for path, patterns in where.manifests.items(): 

236 content = _read(root, path) 

237 for pattern in patterns: 

238 content = pattern.sub(rf"\g<1>{version}\g<3>", content, count=1) 

239 planned[path] = content 

240 if where.images: 

241 compose = _read(root, where.compose) 

242 lines = compose.split("\n") 

243 for n, pin in _pins(compose, where): 

244 lines[n] = f"{pin.group(1)}{line(version)}{pin.group(3)}" 

245 planned[where.compose] = "\n".join(lines) 

246 for path, content in planned.items(): 

247 (root / path).write_text(content, encoding="utf-8") 

248 check(root, f"v{version}", where)

251def notes(root: Path, version: str) -> str: 

252 """`version`'s changelog section, without its heading.""" 

253 text = _read(root, CHANGELOG) 

254 match = re.search(rf"(?m)^## {re.escape(version)} — \d{{4}}-\d{{2}}-\d{{2}}$", text) 

255 if match is None: 

256 raise VersionError(f"{CHANGELOG} has no dated section for {version}") 

257 return _section(text, match.group(0)).strip() + "\n"

260def _section(text: str, heading: str) -> str: 

261 """The text under `heading`, up to the next section's.""" 

262 after = text.split(heading + "\n", 1)[1] if heading + "\n" in text else "" 

263 return re.split(r"(?m)^## ", after, maxsplit=1)[0]

266def main(argv: list[str] | None = None) -> int: 

267 """Run one command; a VersionError is reported and exits 1.""" 

268 parser = argparse.ArgumentParser(description=(__doc__ or "").split("\n", 1)[0]) 

269 parser.add_argument("--root", type=Path, default=Path.cwd()) 

270 commands = parser.add_subparsers(dest="command", required=True) 

271 commands.add_parser("check").add_argument("tag", nargs="?") 

272 commands.add_parser("set").add_argument("version") 

273 commands.add_parser("notes").add_argument("version") 

274 args = parser.parse_args(argv) 

275 try: 

276 if args.command == "check": 

277 print(f"every manifest names {check(args.root, args.tag or None)}") 

278 elif args.command == "set": 

279 set_version(args.root, args.version, date.today()) 

280 else: 

281 sys.stdout.write(notes(args.root, args.version)) 

282 except VersionError as error: 

283 print(f"version: {error}", file=sys.stderr) 

284 return 1 

285 return 0
